5use crate::services::authorization::authorize;
6use crate::types::error_response::{ErrorCode, ErrorResponse, JsonError};
7use anyhow::{anyhow, Result};
8use deadpool_postgres::Client as DbClient;
9use rocket::http::Status;
10use rocket::serde::json::Json;
11use sequent_core::services::jwt::JwtClaims;
12use sequent_core::types::permissions::Permissions;
13use sequent_core::types::scheduled_event::EventProcessors;
14use serde::{Deserialize, Serialize};
15use tracing::instrument;
16use windmill::services::database::get_hasura_pool;
17use windmill::services::{election_dates, election_event_dates};
18
19#[derive(Deserialize, Debug)]
20pub struct ManageElectionDatesBody {
21    election_event_id: String,
22    election_id: Option<String>,
23    scheduled_date: Option<String>,
24    event_processor: EventProcessors,
25}
26
27#[derive(Serialize, Deserialize, Debug)]
28pub struct ManageElectionDatesResponse {
29    error_msg: Option<String>,
30}
31
32#[instrument(skip(claims))]
33#[post("/manage-election-dates", format = "json", data = "<body>")]
34pub async fn manage_election_dates(
35    body: Json<ManageElectionDatesBody>,
36    claims: JwtClaims,
37) -> Result<Json<ManageElectionDatesResponse>, JsonError> {
38    authorize(
39        &claims,
40        true,
41        Some(claims.hasura_claims.tenant_id.clone()),
42        vec![Permissions::SCHEDULED_EVENT_WRITE],
43    )
44    .map_err(|e| {
45        ErrorResponse::new(
46            Status::Unauthorized,
47            &format!("{e:?}"),
48            ErrorCode::Unauthorized,
49        )
50    })?;
51    let input = body.into_inner();
52
53    if input.event_processor == EventProcessors::CREATE_REPORT
54        || input.event_processor == EventProcessors::SEND_TEMPLATE
55    {
56        return Err(ErrorResponse::new(
57            Status::BadRequest,
58            &format!("Invalid event_processors: {:?}", input.event_processor),
59            ErrorCode::InvalidEventProcessor,
60        ));
61    }
62
63    let mut hasura_db_client: DbClient =
64        get_hasura_pool().await.get().await.map_err(|e| {
65            ErrorResponse::new(
66                Status::InternalServerError,
67                &format!("hasura db client failed: {e:?}"),
68                ErrorCode::InternalServerError,
69            )
70        })?;
71    let hasura_transaction =
72        hasura_db_client.transaction().await.map_err(|e| {
73            ErrorResponse::new(
74                Status::InternalServerError,
75                &format!("hasura transaction failed: {e:?}"),
76                ErrorCode::InternalServerError,
77            )
78        })?;
79
80    match input.election_id {
81        Some(id) => {
82            match election_dates::manage_dates(
83                &hasura_transaction,
84                &claims.hasura_claims.tenant_id,
85                &input.election_event_id,
86                &id,
87                input.scheduled_date.as_deref(),
88                input.event_processor.to_string().as_str(),
89            )
90            .await
91            {
92                Ok(_) => (),
93                Err(err) => {
94                    return Ok(Json(ManageElectionDatesResponse {
95                        error_msg: Some(err.to_string()),
96                    }));
97                }
98            }
99        }
100        None => {
101            election_event_dates::manage_dates(
102                &hasura_transaction,
103                &claims.hasura_claims.tenant_id,
104                &input.election_event_id,
105                input.scheduled_date.as_deref(),
106                input.event_processor.to_string().as_str(),
107            )
108            .await
109            .map_err(|e| {
110                ErrorResponse::new(
111                    Status::InternalServerError,
112                    &format!("manage election event dates failed: {e:?}"),
113                    ErrorCode::InternalServerError,
114                )
115            })?;
116        }
117    }
118
119    let _commit = hasura_transaction.commit().await.map_err(|e| {
120        ErrorResponse::new(
121            Status::InternalServerError,
122            &format!("commit failed: {e:?}"),
123            ErrorCode::InternalServerError,
124        )
125    })?;
126
127    Ok(Json(ManageElectionDatesResponse { error_msg: None }))
128}
