21#[derive(Serialize, Deserialize, Debug)]
22pub struct ExportElectionEventInput {
23    election_event_id: String,
24    export_configurations: ExportOptions,
25}
26
27#[derive(Serialize, Deserialize, Debug)]
28pub struct ExportElectionEventOutput {
29    document_id: String,
30    password: Option<String>,
31    task_execution: TasksExecution,
32}
33
34#[instrument(skip(claims))]
35#[post("/export-election-event", format = "json", data = "<input>")]
36pub async fn export_election_event_route(
37    claims: jwt::JwtClaims,
38    input: Json<ExportElectionEventInput>,
39) -> Result<Json<ExportElectionEventOutput>, (Status, String)> {
40    authorize(
41        &claims,
42        true,
43        Some(claims.hasura_claims.tenant_id.clone()),
44        vec![Permissions::ELECTION_EVENT_READ],
45    )?;
46
47    let body = input.into_inner();
48    let tenant_id = claims.hasura_claims.tenant_id.clone();
49    let election_event_id = body.election_event_id.clone();
50    let executer_name = claims
51        .name
52        .clone()
53        .unwrap_or_else(|| claims.hasura_claims.user_id.clone());
54    let mut export_config = body.export_configurations.clone();
55
56    // Insert the task execution record
57    let task_execution = post(
58        &tenant_id,
59        Some(&election_event_id),
60        ETasksExecution::EXPORT_ELECTION_EVENT,
61        &executer_name,
62    )
63    .await
64    .map_err(|error| {
65        (
66            Status::InternalServerError,
67            format!("Failed to insert task execution record: {error:?}"),
68        )
69    })?;
70
71    let document_id = Uuid::new_v4().to_string();
72    let celery_app = get_celery_app().await;
73
74    // todo: generarate only if encrypted
75    let charset: String =
76        "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789-_."
77            .into();
78    let password: Option<String> = if export_config.is_encrypted
79        || export_config.bulletin_board
80        || export_config.reports
81        || export_config.applications
82    {
83        Some(password::generate_random_string_with_charset(64, &charset))
84    } else {
85        None
86    };
87    export_config.password = password.clone();
88
89    let celery_task = celery_app
90        .send_task(export_election_event::export_election_event::new(
91            tenant_id,
92            election_event_id,
93            export_config,
94            document_id.clone(),
95            task_execution.clone(),
96        ))
97        .await
98        .map_err(|error| {
99            (
100                Status::InternalServerError,
101                format!("Error sending export_election_event task: {error:?}"),
102            )
103        })?;
104
105    let output = ExportElectionEventOutput {
106        document_id,
107        password,
108        task_execution: task_execution.clone(),
109    };
110
111    info!("Sent EXPORT_ELECTION_EVENT task  {:?}", &task_execution);
112
113    Ok(Json(output))
114}
