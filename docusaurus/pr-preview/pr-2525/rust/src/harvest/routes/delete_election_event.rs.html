22#[derive(Serialize, Deserialize, Debug)]
23pub struct DeleteElectionEventOutput {
24    id: String,
25    task_execution: TasksExecution,
26    error_msg: Option<String>,
27}
28
29#[derive(Serialize, Deserialize, Debug)]
30pub struct DeleteElectionEventInput {
31    election_event_id: String,
32}
33
34#[instrument(skip(claims))]
35#[post("/delete-election-event", format = "json", data = "<body>")]
36pub async fn delete_election_event_f(
37    body: Json<DeleteElectionEventInput>,
38    claims: JwtClaims,
39) -> Result<Json<DeleteElectionEventOutput>, (Status, String)> {
40    let input = body.into_inner();
41    let tenant_id = claims.hasura_claims.tenant_id.clone();
42    let executer_name = claims
43        .name
44        .clone()
45        .unwrap_or_else(|| claims.hasura_claims.user_id.clone());
46
47    let task_execution = post(
48        &tenant_id,
49        None,
50        ETasksExecution::DELETE_ELECTION_EVENT,
51        &executer_name,
52    )
53    .await
54    .map_err(|error| {
55        (
56            Status::InternalServerError,
57            format!("Failed to insert task execution record: {error:?}"),
58        )
59    })?;
60
61    if let Err(error) = authorize(
62        &claims,
63        true,
64        Some(tenant_id.clone()),
65        vec![Permissions::ELECTION_EVENT_DELETE],
66    ) {
67        let _ = update_fail(
68            &task_execution,
69            &format!("Failed to authorize executing the task: {error:?}"),
70        )
71        .await;
72        return Err(error);
73    };
74
75    let celery_app = get_celery_app().await;
76
77    let realm = get_event_realm(&tenant_id, &input.election_event_id);
78
79    let celery_task_result = celery_app
80        .send_task(delete_election_event::delete_election_event_t::new(
81            claims.hasura_claims.tenant_id.clone(),
82            input.election_event_id.clone(),
83            realm.clone(),
84            task_execution.clone(),
85        ))
86        .await;
87
88    let _celery_task = match celery_task_result {
89        Ok(task) => task,
90        Err(error) => {
91            return Ok(Json(DeleteElectionEventOutput {
92                id: input.election_event_id,
93                error_msg: Some(format!(
94                    "Error sending Delete Election Event task: ${error}"
95                )),
96                task_execution: task_execution.clone(),
97            }));
98        }
99    };
100
101    Ok(Json(DeleteElectionEventOutput {
102        id: input.election_event_id,
103        error_msg: None,
104        task_execution,
105    }))
106}
