5use anyhow::Result;
6use clap::Args;
7use colored::Colorize;
8use sequent_core::services::reports::render_template_text;
9use sequent_core::types::to_map::ToMap;
10use serde_json::{Map, Value};
11use std::fs;
12use windmill::services::reports::manual_verification;
13
14#[derive(clap::ValueEnum, Clone)]
15pub enum TemplateType {
16    Custom,
17    ManualVerification,
18}
19
20/// Render a handlebars-rs template with variables
21#[derive(Args)]
22#[command(about)]
23pub struct RenderTemplate {
24    /// Path to the user handlebars-rs template
25    #[arg(short = 't', long, value_name = "USER_TEMPLATE")]
26    user_template: String,
27
28    /// Path to the user file variables to use in JSON format
29    #[arg(short = 'v', long, value_name = "USER_VARS")]
30    user_vars: String,
31
32    /// Path to the system template
33    #[arg(short = 'T', long, value_name = "SYSTEM_TEMPLATE")]
34    system_template: Option<String>,
35
36    /// Path to the system file variables to use in JSON format
37    #[arg(short = 'V', long, value_name = "SYSTEM_VARS")]
38    system_vars: Option<String>,
39
40    /// Type of the user template
41    #[arg(
42        short = 'y',
43        long,
44        value_name = "TEMPLATE_TYPE",
45        default_value = "custom"
46    )]
47    template_type: TemplateType,
48
49    /// Path to the output HTML file
50    #[arg(short = 'o', long, value_name = "OUTPUT")]
51    output: String,
52}
53
54impl RenderTemplate {
55    /// Execute the rendering process
56    pub fn run(&self) {
57        match self.generate_report() {
58            Ok(_) => println!("{}", "Successfully generated the report".green()),
59            Err(err) => eprintln!("Error! Failed to generate the report: {err:?}"),
60        }
61    }
62
63    /// Generate the report by reading templates, parsing variables, and
64    /// rendering the output
65    fn generate_report(&self) -> Result<(), Box<dyn std::error::Error>> {
66        let user_vars_content = fs::read_to_string(&self.user_vars)
67            .map_err(|e| format!("Could not read user variables file: {e:?}"))?;
68
69        // Parse the user or system template variables based on the template type
70        let user_vars: Map<String, Value> = match self.template_type {
71            TemplateType::Custom => {
72                let vars_json: Map<String, Value> = serde_json::from_str(&user_vars_content)
73                    .map_err(|e| format!("Could not parse user template variables: {e:?}"))?;
74                vars_json
75            }
76            TemplateType::ManualVerification => {
77                let system_vars_content = fs::read_to_string(&self.user_vars)
78                    .map_err(|e| format!("Could not read system variables file: {e:?}"))?;
79                let system_template_data: manual_verification::UserData =
80                    serde_json::from_str(&system_vars_content)
81                        .map_err(|e| format!("Could not parse system template variables: {e:?}"))?;
82                system_template_data.to_map()?
83            }
84        };
85
86        // Load the appropriate user or manual verification template based on
87        // the template type
88        let user_template = match self.template_type {
89            TemplateType::Custom => fs::read_to_string(&self.user_template)
90                .map_err(|e| format!("Could not read user template file: {e:?}"))?,
91            TemplateType::ManualVerification => {
92                let custom_template = self.user_template.clone();
93                fs::read_to_string(custom_template).map_err(|e| {
94                    format!("Could not read manual verification template file: {e:?}")
95                })?
96            }
97        };
98
99        let rendered_output = self.render_template_with_vars(user_template, user_vars)?;
100
101        fs::write(&self.output, rendered_output)
102            .map_err(|e| format!("Failed to write the output file: {e:?}"))?;
103
104        Ok(())
105    }
106
107    /// Render the template with the provided variables and base template
108    fn render_template_with_vars(
109        &self,
110        user_template: String,
111        user_vars: Map<String, Value>,
112    ) -> Result<String, String> {
113        let rendered_user_template: String = render_template_text(&user_template, user_vars)
114            .map_err(|e| format!("User template rendering error: {e:?}"))?;
115
116        // Determine the system template content based on the template type
117        let system_template = match &self.template_type {
118            TemplateType::Custom => "{{{rendered_user_template}}}".to_string(),
119            TemplateType::ManualVerification => {
120                if let Some(system_template) = &self.system_template {
121                    fs::read_to_string(system_template).map_err(|e| {
122                        format!(
123                            "Could not read manual verification system template file {system_template:?}: {e:?}",
124                        )
125                    })?
126                } else {
127                    include_str!("../../../../.devcontainer/minio/public-assets/manual_verification_system.hbs").to_string()
128                }
129            }
130        };
131
132        let mut system_vars: Map<String, Value> = match self.template_type {
133            TemplateType::Custom => Default::default(),
134            TemplateType::ManualVerification => {
135                let vars_path: String = self
136                    .system_vars
137                    .clone()
138                    .ok_or(format!("System vars not provided"))?;
139                let system_vars_content = fs::read_to_string(&vars_path)
140                    .map_err(|e| format!("Could not read system variables file: {e:?}"))?;
141                let system_template_data: manual_verification::SystemData =
142                    serde_json::from_str(&system_vars_content)
143                        .map_err(|e| format!("Could not parse system template variables: {e:?}"))?;
144                system_template_data
145                    .to_map()
146                    .map_err(|e| format!("Error converting into map: {e:?}"))?
147            }
148        };
149        system_vars.insert(
150            "rendered_user_template".to_string(),
151            Value::String(rendered_user_template),
152        );
153
154        render_template_text(&system_template, system_vars)
155            .map_err(|e| format!("System template rendering error: {e:?}"))
156    }
157}
