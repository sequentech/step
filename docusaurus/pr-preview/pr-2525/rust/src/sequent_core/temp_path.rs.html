51/// Gets the public assets path from the environment variable.
52///
53/// # Errors
54/// Returns an error if the environment variable is not set or invalid.
55pub fn get_public_assets_path_env_var() -> Result<String> {
56    match env::var("PUBLIC_ASSETS_PATH") {
57        Ok(path) => Ok(path),
58        Err(e) => Err(e)
59            .with_context(|| "Error fetching PUBLIC_ASSETS_PATH env var")?,
60    }
61}
62/// Gets the file size for the given file path.
63///
64/// # Errors
65/// Returns an error if the file does not exist or cannot be accessed.
66pub fn get_file_size(filepath: &str) -> Result<u64> {
67    let metadata = fs::metadata(filepath)?;
68    Ok(metadata.len())
69}

71/*
72 * Writes data into a named temp file. The temp file will have the
73 * specificed prefix and suffix.
74 *
75 * Returns the TempPath of the file, the stringified version of the path to
76 * the file and the bytes size of the file.
77 *
78 * NOTE: The file will be dropped when the TempPath goes out of the scope.
79 * Returning the TempPath, even if the variable goes unused, allows the
80 * caller to control the lifetime of the created temp file.
81 */
82#[instrument(skip(data), err)]
83/// # Errors
84/// Returns an error if the file cannot be created or written.
85pub fn write_into_named_temp_file(
86    data: &[u8],
87    prefix: &str,
88    suffix: &str,
89) -> Result<(TempPath, String, u64)> {
90    let file: NamedTempFile = generate_temp_file(prefix, suffix)
91        .with_context(|| "Error creating named temp file")?;
92    {
93        let file2 = file
94            .reopen()
95            .with_context(|| "Couldn't reopen file for writing")?;
96        let mut buf_writer = BufWriter::new(file2);
97        buf_writer
98            .write(data)
99            .with_context(|| "Error writing into named temp file")?;
100        buf_writer
101            .flush()
102            .with_context(|| "Error calling flush into named temp file")?;
103    }
104    let temp_path = file.into_temp_path();
105    let temp_path_string = temp_path.to_string_lossy().to_string();
106    let file_size = get_file_size(temp_path_string.as_str())
107        .with_context(|| "Error obtaining file size")?;
108    Ok((temp_path, temp_path_string, file_size))
109}

112/// Generates a named temporary file with the given prefix and suffix.
113///
114/// # Errors
115/// Returns an error if the file cannot be created.
116pub fn generate_temp_file(prefix: &str, suffix: &str) -> Result<NamedTempFile> {
117    // Get the system's temporary directory.
118    let temp_dir = env::temp_dir();
119
120    // Use the Builder to create a temporary file within the temporary
121    // directory. The file will be deleted when the `NamedTempFile` object
122    // goes out of scope.
123    let temp_file = Builder::new()
124        .prefix(prefix) // Optional: specify a prefix for the file name.
125        .suffix(suffix) // Optional: specify a suffix for the file name.
126        .rand_bytes(12) // Optional: specify the number of random bytes to use for the name.
127        .tempfile_in(&temp_dir)
128        .with_context(|| "Error generating temp file")?;
129
130    Ok(temp_file)
131}

134/// Reads the contents of a named temporary file.
135///
136/// # Errors
137/// Returns an error if the file cannot be read.
138pub fn read_temp_file(temp_file: &mut NamedTempFile) -> Result<Vec<u8>> {
139    // Rewind the file to the beginning to read its contents
140    temp_file.rewind()?;
141
142    // Read the file's contents into a Vec<u8>
143    let mut file_bytes = Vec::new();
144    temp_file.read_to_end(&mut file_bytes)?;
145    Ok(file_bytes)
146}

148#[instrument(err)]
149/// Reads the contents of a temporary file path.
150///
151/// # Errors
152/// Returns an error if the file cannot be read.
153pub fn read_temp_path(temp_path: &TempPath) -> Result<Vec<u8>> {
154    let mut file = File::open(temp_path)?;
155    let mut buffer = Vec::new();
156    file.read_to_end(&mut buffer)?;
157    Ok(buffer)
