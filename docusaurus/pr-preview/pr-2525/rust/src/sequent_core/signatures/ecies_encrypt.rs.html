29#[instrument(skip(password), err)]
30/// Encrypts a string using ECIES with the given public key and password.
31///
32/// # Errors
33/// Returns an error if encryption fails or file operations fail.
34pub fn ecies_encrypt_string(
35    public_key_pem: &str,
36    password: &str,
37) -> Result<String> {
38    let temp_pem_file = generate_temp_file("public_key", ".pem")?;
39    let temp_pem_file_path = temp_pem_file.path();
40    let temp_pem_file_string = temp_pem_file_path.to_string_lossy().to_string();
41    // Write the salt and encrypted data to the output file
42    // Using brackets: let it drop out of scope so that all bytes are written
43    {
44        let mut output_file = File::create(temp_pem_file_path)
45            .context("Failed to create file")?;
46        output_file
47            .write_all(public_key_pem.as_bytes())
48            .context("Failed to write file")?;
49    }
50    // Encode the &[u8] to a Base64 string
51
52    let command = format!(
53        "java -jar {ECIES_TOOL_PATH} encrypt {temp_pem_file_string} {password}"
54    );
55    info!("command: '{command}'");
56
57    let result = run_shell_command(&command)?.replace('\n', "");
58
59    info!("ecies_encrypt_string: '{result}'");
60
61    Ok(result)
62}

64#[instrument(err)]
65/// Generates an ECIES key pair.
66///
67/// # Errors
68/// Returns an error if key generation fails or file operations fail.
69pub fn generate_ecies_key_pair() -> Result<EciesKeyPair> {
70    let temp_private_pem_file = generate_temp_file("private_key", ".pem")?;
71    let temp_private_pem_file_path = temp_private_pem_file.path();
72    let temp_private_pem_file_string =
73        temp_private_pem_file_path.to_string_lossy().to_string();
74
75    let temp_public_pem_file = generate_temp_file("public_key", ".pem")?;
76    let temp_public_pem_file_path = temp_public_pem_file.path();
77    let temp_public_pem_file_string =
78        temp_public_pem_file_path.to_string_lossy().to_string();
79
80    let command = format!(
81        "java -jar {ECIES_TOOL_PATH} create-keys {temp_public_pem_file_string} {temp_private_pem_file_string}"
82    );
83    run_shell_command(&command)?;
84
85    let private_key_pem = fs::read_to_string(temp_private_pem_file_path)?;
86    let public_key_pem = fs::read_to_string(temp_public_pem_file_string)?;
87
88    info!("generate_ecies_key_pair(): public_key_pem: {public_key_pem:?}");
89
90    Ok(EciesKeyPair {
91        private_key_pem,
92        public_key_pem,
93    })
94}

97/// Signs data using ECIES.
98///
99/// # Errors
100/// Returns an error if signing fails or file operations fail.
101pub fn ecies_sign_data(
102    acm_key_pair: &EciesKeyPair,
103    data: &str,
104) -> Result<String> {
105    // Retrieve the PEM as a string
106    info!("pem: {}", acm_key_pair.private_key_pem);
107
108    let temp_pem_file = generate_temp_file("private_key", ".pem")?;
109    let temp_pem_file_path = temp_pem_file.path();
110    let temp_pem_file_string = temp_pem_file_path.to_string_lossy().to_string();
111    // Write the salt and encrypted data to the output file
112    // Using brackets: let it drop out of scope so that all bytes are written
113    {
114        let mut output_file = File::create(temp_pem_file_path)
115            .context("Failed to create file")?;
116        output_file
117            .write_all(acm_key_pair.private_key_pem.as_bytes())
118            .context("Failed to write file")?;
119    }
120    let temp_data_file = generate_temp_file("data", ".eml")?;
121    let temp_data_file_path = temp_data_file.path();
122    let temp_data_file_string =
123        temp_data_file_path.to_string_lossy().to_string();
124    // Write the salt and encrypted data to the output file
125    {
126        let mut output_file = File::create(temp_data_file_path)
127            .context("Failed to create file")?;
128        output_file
129            .write_all(data.as_bytes())
130            .context("Failed to write file")?;
131    }
132
133    let command = format!(
134        "java -jar {ECIES_TOOL_PATH} sign {temp_pem_file_string} {temp_data_file_string}"
135    );
136
137    let encrypted_base64 = run_shell_command(&command)?.replace('\n', "");
138
139    info!("ecies_sign_data: '{encrypted_base64}'");
140
141    Ok(encrypted_base64)
142}

152#[instrument(skip_all, err)]
153/// Signs multiple data items using ECIES in bulk.
154///
155/// # Errors
156/// Returns an error if signing fails or file operations fail.
157pub fn ecies_sign_data_bulk(
158    acm_key_pair: &EciesKeyPair,
159    requests: &[SignRequest],
160) -> Result<HashMap<String, String>> {
161    // If there are no requests, just return an empty map
162    if requests.is_empty() {
163        return Ok(HashMap::new());
164    }
165
166    // 1. Create a temporary directory (folder) for bulk-signing
167    let tmp_dir = tempdir().context("Failed to create temporary directory")?;
168
169    // 2. Write the private key into that directory, e.g. private_key.pem
170    let private_key_path = tmp_dir.path().join("private_key.pem");
171    {
172        let mut key_file = File::create(&private_key_path)
173            .context("Failed to create private_key.pem")?;
174        key_file
175            .write_all(acm_key_pair.private_key_pem.as_bytes())
176            .context("Failed to write private_key.pem")?;
177    }
178
179    // 3. For each request, create a file with the sign_data content E.g.
180    //    "sign_0001.txt", "sign_0002.txt", etc. We'll store a small structure
181    //    to track (id -> filename).
182    let mut file_map: HashMap<String, PathBuf> = HashMap::new();
183    for (i, req) in requests.iter().enumerate() {
184        let filename = format!("sign_{i:04}.txt");
185        let path = tmp_dir.path().join(&filename);
186
187        {
188            let mut f = File::create(&path)
189                .with_context(|| format!("Failed to create {filename}"))?;
190            f.write_all(req.data.as_bytes())
191                .with_context(|| format!("Failed to write {filename}"))?;
192        }
193
194        file_map.insert(req.id.clone(), path);
195    }
196
197    // 4. Build the sign-bulk command (one call). sign-bulk <private_key_file>
198    //    <folder_to_sign>
199    //
200    //    The second parameter is just the directory path;
201    //    the Java tool will sign every file that doesn't end with .sign
202    let cmd = format!(
203        "java -jar {ecies_tool_path} sign-bulk {key} {folder}",
204        ecies_tool_path = ECIES_TOOL_PATH,
205        key = private_key_path.to_string_lossy(),
206        folder = tmp_dir.path().to_string_lossy(),
207    );
208    info!("Running sign-bulk => {cmd}");
209
210    // 5. Execute the shell command (similar to your run_shell_command).
211    run_shell_command(&cmd)?;
212
213    // 6. After sign-bulk finishes, we collect the results: For each
214    //    sign_xxxx.txt, the tool should have produced sign_xxxx.txt.sign We'll
215    //    read them into a map of (id -> signature_base64)
216    let mut signature_map = HashMap::new();
217    for (id, path) in &file_map {
218        // the Java tool will create the file with .sign appended
219        let sign_file = path.with_extension("txt.sign");
220        if !sign_file.exists() {
221            return Err(anyhow::anyhow!(
222                "Expected signature file not found: {}",
223                sign_file.display()
224            ));
225        }
226        let signature_b64 = std::fs::read_to_string(&sign_file)
227            .with_context(|| {
228                format!("Failed to read signature file {}", sign_file.display())
229            })?
230            .trim()
231            .to_string();
232
233        signature_map.insert(id.clone(), signature_b64);
234    }
235
236    // 7. Return all signatures in a HashMap keyed by the "id"
237    Ok(signature_map)
238}
