4use regex::Regex;
5use std::collections::HashMap;
6use std::collections::HashSet;
7use std::env;
8use tracing::instrument;
9use uuid::Uuid;
10
11#[instrument(skip(input))]
12/// Replaces UUIDs in the input string, except those in the keep list or fixed config.
13///
14/// # Panics
15/// Panics if the UUID regex is invalid or if a capture group is missing (should not happen).
16#[allow(clippy::unwrap_used)] // Safe: regex is a static, known-valid constant; failure would indicate a programming error.
17pub fn replace_uuids(
18    input: &str,
19    keep: &[String],
20) -> (String, HashMap<String, String>) {
21    let uuid_regex =
22        Regex::new(r"[0-9a-fA-F]{8}(-[0-9a-fA-F]{4}){3}-[0-9a-fA-F]{12}")
23            .expect("Invalid UUID regex");
24    let fixed_uuids_from_config = env::var("ELECTION_EVENT_FIXED_UUIDS")
25        .unwrap_or_default()
26        .split(',')
27        .map(std::string::ToString::to_string)
28        .collect::<Vec<String>>();
29    let mut keep_all = keep.to_vec();
30    keep_all.extend(fixed_uuids_from_config);
31
32    let mut seen_uuids = HashMap::new();
33    let keep_set: HashSet<String> = keep_all.into_iter().collect();
34
35    let result = uuid_regex
36        .replace_all(input, |caps: &regex::Captures| {
37            let old_uuid = caps
38                .get(0)
39                .expect("capture group 0 should always be present")
40                .as_str()
41                .to_string();
42            if keep_set.contains(&old_uuid) {
43                old_uuid.clone()
44            } else {
45                seen_uuids
46                    .entry(old_uuid.clone())
47                    .or_insert_with(|| Uuid::new_v4().to_string())
48                    .clone()
49            }
50        })
51        .into_owned();
52    (result, seen_uuids)
53}
