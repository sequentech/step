10/// A tree node that corresponds to an area
11#[derive(PartialEq, Eq, Debug, Clone, Serialize, Deserialize)]
12pub struct TreeNodeArea {
13    pub id: String, // area id
14    pub tenant_id: String,
15    pub annotations: Option<Value>,
16    pub election_event_id: String,
17    pub parent_id: Option<String>,
18}
19
20// Extra data for an area. We'll use that to create a tree
21// where all nodes have in "contest_ids" both their directly assigned
22// contests and the contests inherited from their ancestors.
23#[derive(PartialEq, Eq, Debug, Clone, Default, Serialize, Deserialize)]
24pub struct ContestsData {
25    /// Set of contest IDs associated with the area
26    contest_ids: HashSet<String>,
27}
28
29impl From<&Area> for TreeNodeArea {
30    fn from(area: &Area) -> Self {
31        TreeNodeArea {
32            id: area.id.clone(),
33            tenant_id: area.tenant_id.clone(),
34            annotations: area.annotations.clone(),
35            election_event_id: area.election_event_id.clone(),
36            parent_id: area.parent_id.clone(),
37        }
38    }
39}

41#[derive(PartialEq, Eq, Debug, Clone, Serialize, Deserialize)]
42pub struct TreeNode<T = ()> {
43    pub area: Option<TreeNodeArea>,
44    pub children: Vec<TreeNode<T>>,
45    pub data: T,
46}
47
48impl<T> TreeNode<T>
49where
50    T: Clone + Default,
51{
52    /// Returns all nodes in the tree as a flat vector of `TreeNodeArea`.
53    #[must_use]
54    pub fn get_all_children(&self) -> Vec<TreeNodeArea> {
55        let mut children: Vec<TreeNodeArea> = vec![];
56        if let Some(area) = self.area.clone() {
57            children.push(area);
58        }
59        let sub_children: Vec<TreeNodeArea> = self
60            .children
61            .iter()
62            .flat_map(TreeNode::get_all_children)
63            .collect();
64        children.extend(sub_children);
65        children
66    }
67
68    /// Creates a tree from the list of nodes.
69    ///
70    /// # Errors
71    /// Returns an error if a parent id is not found or a loop is detected.
72    pub fn from_areas(areas: Vec<TreeNodeArea>) -> Result<TreeNode<T>> {
73        let mut nodes: HashMap<String, TreeNode<T>> = HashMap::new();
74        let mut parent_map: HashMap<String, Vec<String>> = HashMap::new();
75        let mut root_ids: Vec<String> = Vec::new();
76
77        // Initialize TreeNodes and parent map
78        for area in areas {
79            let id = area.id.clone();
80            let parent_id = area.parent_id.clone();
81
82            nodes.insert(
83                id.clone(),
84                TreeNode::<T> {
85                    area: Some(area),
86                    children: Vec::new(),
87                    data: Default::default(), // this should change?
88                },
89            );
90
91            if let Some(parent_id) = parent_id {
92                parent_map.entry(parent_id).or_default().push(id);
93            } else {
94                root_ids.push(id);
95            }
96        }
97
98        // Ensure all parent_ids are valid
99        for parent_id in parent_map.keys() {
100            if !nodes.contains_key(parent_id) {
101                return Err(anyhow!(
102                    "Parent id {parent_id} not found in the tree structure"
103                ));
104            }
105        }
106
107        let mut root_node = TreeNode::<T> {
108            area: None,
109            children: Vec::new(),
110            data: Default::default(),
111        };
112
113        // Build the forest under a single root
114        // as build_tree is recursive, we defined the visited var outside to
115        // maintain state outside the multiple recursive calls
116        let mut visited: HashSet<String> = HashSet::new();
117        for root_id in root_ids {
118            let child_node = TreeNode::<T>::build_tree(
119                &root_id,
120                &nodes,
121                &mut visited,
122                &parent_map,
123            )?;
124            root_node.children.push(child_node);
125        }
126
127        Ok(root_node)
128    }
129
130    /// Internal function used by `from_areas` to recursively build the tree.
131    ///
132    /// # Errors
133    /// Returns an error if a loop is detected or a node is not found.
134    fn build_tree<'a>(
135        id: &'a str,
136        nodes: &'a HashMap<String, TreeNode<T>>,
137        visited: &mut HashSet<String>,
138        parent_map: &'a HashMap<String, Vec<String>>,
139    ) -> Result<TreeNode<T>> {
140        if visited.contains(id) {
141            return Err(anyhow!("Loop detected in the tree structure"));
142        }
143
144        visited.insert(id.to_string());
145        let node = nodes
146            .get(id)
147            .ok_or_else(|| anyhow!("Node not found"))?
148            .clone();
149        let mut new_node = TreeNode::<T> {
150            area: node.area.clone(),
151            children: Vec::new(),
152            data: Default::default(), // this should change?
153        };
154
155        if let Some(children_ids) = parent_map.get(id) {
156            for child_id in children_ids {
157                let child_node = TreeNode::<T>::build_tree(
158                    child_id, nodes, visited, parent_map,
159                )?;
160                new_node.children.push(child_node);
161            }
162        }
163
164        visited.remove(id);
165        Ok(new_node)
166    }

168    /// Finds an area in the tree by its id.
169    #[must_use]
170    pub fn find_area(&self, area_id: &str) -> Option<TreeNode<T>> {
171        if let Some(area) = self.area.clone() {
172            if area.id == area_id {
173                return Some(self.clone());
174            }
175        }
176        for leave in &self.children {
177            if let Some(area) = leave.find_area(area_id) {
178                return Some(area);
179            }
180        }
181        None
182    }
183
184    /// Finds the path from the root to the area with the given id.
185    #[must_use]
186    pub fn find_path_to_area(
187        &self,
188        area_id: &str,
189    ) -> Option<Vec<TreeNodeArea>> {
190        // Initialize the path and call the helper function
191        let mut path: Vec<TreeNodeArea> = Vec::new();
192        if Self::dfs(self, area_id, &mut path) {
193            Some(path)
194        } else {
195            None
196        }
197    }
198
199    /// Depth First Helper function to recursively find the path.
200    fn dfs(
201        node: &TreeNode<T>,
202        area_id: &str,
203        path: &mut Vec<TreeNodeArea>,
204    ) -> bool {
205        // Add current node to the path if it has an area
206        if let Some(area) = &node.area {
207            path.push(area.clone());
208        }
209
210        // Check if the current node is the target node
211        if node.area.as_ref().is_some_and(|area| area.id == area_id) {
212            return true;
213        }
214
215        // Recur for each child
216        for child in &node.children {
217            if Self::dfs(child, area_id, path) {
218                return true;
219            }
220        }
221
222        // If not found, backtrack
223        if node.area.is_some() {
224            path.pop();
225        }
226        false
227    }

229    // generate a tree where each node has the data of all related contests
230    // note that areas spread down the tree
231    pub fn get_contests_data_tree(
232        &self,
233        area_contests: &[AreaContest],
234    ) -> TreeNode<ContestsData> {
235        // Map<area_id, Set<contest_id>>
236        let mut areas_map: HashMap<String, HashSet<String>> = HashMap::new();
237        for area_contest in area_contests {
238            areas_map
239                .entry(area_contest.area_id.clone())
240                .and_modify(|contest_ids| {
241                    contest_ids.insert(area_contest.contest_id.clone());
242                })
243                .or_insert_with(|| {
244                    let mut set = HashSet::new();
245                    set.insert(area_contest.contest_id.clone());
246                    set
247                });
248        }
249        let root_data: ContestsData = ContestsData::default();
250        self.contests_data_tree(&root_data, &areas_map)
251    }
252
253    /// Recursively builds a tree of contest data for each node.
254    fn contests_data_tree(
255        &self,
256        parent_data: &ContestsData,
257        // Map<area_id, Set<contest_id>>
258        areas_map: &HashMap<String, HashSet<String>>,
259    ) -> TreeNode<ContestsData> {
260        // get contest ids inherited from parent
261        let mut contest_ids = parent_data.contest_ids.clone();
262        // get contest ids from self area
263        if let Some(area) = self.area.clone() {
264            if let Some(self_contests) = areas_map.get(&area.id) {
265                contest_ids.extend(self_contests.clone());
266            }
267        }
268        let data = ContestsData { contest_ids };
269        let children: Vec<TreeNode<ContestsData>> = self
270            .children
271            .iter()
272            .map(|child| {
273                child.contests_data_tree(
274                    &data, // Map<area_id, Set<contest_id>>
275                    areas_map,
276                )
277            })
278            .collect();
279        TreeNode::<ContestsData> {
280            area: self.area.clone(),
281            children,
282            data,
283        }
284    }
285}

287/// Methods for contest matching on contest data trees.
288impl TreeNode<ContestsData> {
289    /// For a given `TreeNode` of type `ContestsData`, return all
290    /// area-contests. Note that this will include indirect/inherited ones.
291    #[must_use]
292    #[allow(clippy::used_underscore_binding)]
293    pub fn get_contest_matches(
294        &self,
295        _contest_ids: &HashSet<String>,
296    ) -> HashSet<AreaContest> {
297        let mut set = HashSet::new();
298        if let Some(area) = self.area.clone() {
299            let own_area_contests: HashSet<AreaContest> = self
300                .data
301                .contest_ids
302                .iter()
303                .map(|contest_id| AreaContest {
304                    id: area.id.clone(),
305                    area_id: area.id.clone(),
306                    contest_id: contest_id.clone(),
307                })
308                .collect();
309            set.extend(own_area_contests);
310        }
311        for child in &self.children {
312            let child_set = child.get_contest_matches(_contest_ids);
313            set.extend(child_set);
314        }
315        set
316    }
317}

319/// Iterator for traversing a tree node structure.
320#[allow(missing_docs)]
321pub struct TreeNodeIter<'a, T> {
322    #[allow(missing_docs, clippy::missing_docs_in_private_items)]
323    queue: VecDeque<&'a TreeNode<T>>,
324}
325
326impl<'a, T> Iterator for TreeNodeIter<'a, T> {
327    type Item = &'a TreeNode<T>;
328
329    fn next(&mut self) -> Option<Self::Item> {
330        if let Some(node) = self.queue.pop_front() {
331            for child in &node.children {
332                self.queue.push_back(child);
333            }
334            Some(node)
335        } else {
336            None
337        }
338    }
339}
340
341impl<T> TreeNode<T> {
342    /// Returns an iterator over the tree nodes.
343    #[must_use]
344    pub fn iter(&self) -> TreeNodeIter<T> {
345        let mut queue = VecDeque::new();
346        queue.push_back(self);
347        TreeNodeIter { queue }
348    }
349}
350
351impl<'a, T> IntoIterator for &'a TreeNode<T> {
352    type Item = &'a TreeNode<T>;
353    type IntoIter = TreeNodeIter<'a, T>;
354
355    fn into_iter(self) -> Self::IntoIter {
356        self.iter()
357    }
358}
