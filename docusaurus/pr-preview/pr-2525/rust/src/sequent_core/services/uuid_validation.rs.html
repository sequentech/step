5use anyhow::anyhow;
6use uuid::Uuid;
7
8/// Parses and validates that the given string is a valid v4 UUID.
9///
10/// Returns the parsed `Uuid` on success, or an error if the string is not
11/// a valid UUID or is not version 4 (random).
12///
13/// # Errors
14/// Returns an error if the string is not a valid UUID or is not version 4.
15pub fn parse_uuid_v4(value: &str) -> anyhow::Result<Uuid> {
16    let uuid = Uuid::parse_str(value)
17        .map_err(|e| anyhow!("invalid UUID '{value}': {e}"))?;
18
19    match uuid.get_version() {
20        Some(uuid::Version::Random) => Ok(uuid),
21        other => Err(anyhow!("UUID '{value}' is not v4 (version: {other:?})")),
22    }
23}
24
25/// Parses and validates that the given string is a valid v4 UUID,
26/// returning a descriptive error that includes the field name.
27///
28/// # Errors
29/// Returns an error if the string is not a valid UUID or is not version 4.
30pub fn parse_uuid_v4_field(
31    value: &str,
32    field_name: &str,
33) -> anyhow::Result<Uuid> {
34    parse_uuid_v4(value)
35        .map_err(|e| anyhow!("invalid v4 UUID for field '{field_name}': {e}"))
36}

38#[cfg(test)]
39mod tests {
40    use super::*;
41
42    #[test]
43    fn test_valid_v4_uuid() {
44        let uuid = Uuid::new_v4().to_string();
45        assert!(parse_uuid_v4(&uuid).is_ok());
46    }
47
48    #[test]
49    fn test_valid_v4_uuid_uppercase() {
50        let uuid = Uuid::new_v4().to_string().to_uppercase();
51        assert!(parse_uuid_v4(&uuid).is_ok());
52    }
53
54    #[test]
55    fn test_invalid_uuid_string() {
56        assert!(parse_uuid_v4("not-a-uuid").is_err());
57    }
58
59    #[test]
60    fn test_empty_string() {
61        assert!(parse_uuid_v4("").is_err());
62    }
63
64    #[test]
65    fn test_nil_uuid_is_not_v4() {
66        assert!(parse_uuid_v4("00000000-0000-0000-0000-000000000000").is_err());
67    }
68
69    #[test]
70    fn test_parse_uuid_v4_field_error_includes_field_name() {
71        let err = parse_uuid_v4_field("bad", "tenant_id").unwrap_err();
72        assert!(err.to_string().contains("tenant_id"));
73    }
74}
