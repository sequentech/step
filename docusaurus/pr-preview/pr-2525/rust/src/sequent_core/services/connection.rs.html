22/// Header for tenant ID
23const TENANT_ID_HEADER: &str = "tenant-id";
24/// Header for event ID
25const EVENT_ID_HEADER: &str = "event-id";
26/// Header for authorization
27const AUTHORIZATION_HEADER: &str = "authorization";
28pub const PRE_EXPIRATION_SECS: i64 = 5;
29/// Authentication headers extracted from a request.
30#[derive(Debug, Clone, Deserialize, Serialize)]
31pub struct AuthHeaders {
32    pub key: String,
33    pub value: String,
34}
35
36#[rocket::async_trait]
37impl<'r> FromRequest<'r> for AuthHeaders {
38    type Error = ();
39
40    async fn from_request(
41        request: &'r Request<'_>,
42    ) -> Outcome<Self, Self::Error> {
43        let headers = request.headers().clone();
44        if let Some(value) = headers.get_one("X-Hasura-Admin-Secret") {
45            Outcome::Success(AuthHeaders {
46                key: "X-Hasura-Admin-Secret".to_string(),
47                value: value.to_string(),
48            })
49        } else if let Some(value) = headers.get_one("authorization") {
50            Outcome::Success(AuthHeaders {
51                key: "authorization".to_string(),
52                value: value.to_string(),
53            })
54        } else {
55            warn!("AuthHeaders guard: headers: {headers:?}");
56            Outcome::Error((Status::Unauthorized, ()))
57        }
58    }
59}

61#[rocket::async_trait]
62impl<'r> FromRequest<'r> for JwtClaims {
63    type Error = ();
64
65    async fn from_request(
66        request: &'r Request<'_>,
67    ) -> Outcome<Self, Self::Error> {
68        let headers = request.headers().clone();
69        if let Some(authorization) = headers.get_one("authorization") {
70            if let Some(token) = authorization.strip_prefix("Bearer ") {
71                match decode_jwt(token) {
72                    Ok(jwt) => Outcome::Success(jwt),
73                    Err(err) => {
74                        warn!("JwtClaims guard: decode_jwt error {err:?}");
75                        Outcome::Error((Status::Unauthorized, ()))
76                    }
77                }
78            } else {
79                warn!("JwtClaims guard: not a bearer token: {authorization:?}");
80                Outcome::Error((Status::Unauthorized, ()))
81            }
82        } else {
83            warn!("JwtClaims guard: headers: {headers:?}");
84            Outcome::Error((Status::Unauthorized, ()))
85        }
86    }
87}

89/// User location information extracted from headers.
90#[derive(Debug)]
91#[allow(missing_docs)]
92pub struct UserLocation {
93    pub ip: Option<IpAddr>,
94    pub country_code: Option<String>,
95}
96
97#[rocket::async_trait]
98impl<'r> FromRequest<'r> for UserLocation {
99    type Error = ();
100
101    async fn from_request(
102        request: &'r Request<'_>,
103    ) -> Outcome<Self, Self::Error> {
104        let ip = request
105            .headers()
106            .get_one("CF-Connecting-IP")
107            .or_else(|| request.headers().get_one("X-Forwarded-For"))
108            .and_then(|ip_str| ip_str.parse().ok());
109
110        let country_code = request
111            .headers()
112            .get_one("CF-IPCountry")
113            .map(str::to_string);
114
115        Outcome::Success(UserLocation { ip, country_code })
116    }
117}

129/// Datafix credentials for authorization.
130#[derive(Debug)]
131#[allow(missing_docs)]
132struct DatafixCredentials {
133    /// Client ID
134    client_id: String,
135    /// Client secret
136    client_secret: String,
137}

139/// Datafix headers extracted from the request.
140#[derive(Debug)]
141#[allow(missing_docs)]
142struct DatafixHeaders {
143    /// Tenant ID
144    tenant_id: String,
145    /// Event ID
146    event_id: String,
147    /// Authorization credentials
148    authorization: DatafixCredentials,
149}
150
151/// Returns None if any of the required headers are missing or is incomplete
152#[instrument(skip_all)]
153fn parse_datafix_headers(headers: &HeaderMap) -> Option<DatafixHeaders> {
154    let required_headers =
155        [TENANT_ID_HEADER, EVENT_ID_HEADER, AUTHORIZATION_HEADER];
156    let mut missing_headers = vec![];
157    for header in required_headers {
158        if !headers.contains(header) {
159            warn!("DatafixClaims guard: No {header} in headers: {headers:?}");
160            missing_headers.push(header);
161        }
162    }
163
164    if !missing_headers.is_empty() {
165        return None;
166    }
167
168    let (tenant_id, event_id, authorization) = (
169        headers.get_one(TENANT_ID_HEADER).unwrap_or_default(),
170        headers.get_one(EVENT_ID_HEADER).unwrap_or_default(),
171        headers.get_one(AUTHORIZATION_HEADER).unwrap_or_default(),
172    );
173
174    info!(
175        "tenant-id: {:?} event-id: {:?} authorization: {:?}",
176        tenant_id, event_id, authorization
177    );
178
179    let mut auth_collection = authorization.split(':');
180    let client_id = auth_collection.next();
181    let client_secret = auth_collection.next();
182    info!("{:?}:{:?}", client_id, client_secret);
183    let (Some(client_id), Some(client_secret)) = (client_id, client_secret)
184    else {
185        return None;
186    };
187
188    Some(DatafixHeaders {
189        tenant_id: tenant_id.to_string(),
190        event_id: event_id.to_string(),
191        authorization: DatafixCredentials {
192            client_id: client_id.to_string(),
193            client_secret: client_secret.to_string(),
194        },
195    })
196}

198/// `TokenResponse`, timestamp before sending the request and the credentials to
199/// make sure the requester is the same.
200#[derive(Debug, Clone)]
201struct TokenResponseExtended {
202    /// Access token response from Keycloak
203    token_resp: PubKeycloakAdminToken,
204    /// Timestamp when the token was requested
205    stamp: Instant,
206    /// Client ID used to request the token
207    client_id: String,
208    /// Client secret used to request the token
209    client_secret: String,
210    /// Tenant ID used to request the token
211    tenant_id: String,
212}
213
214/// Last access token can be reused if it´s not expired, this is to avoid
215/// Keycloak having to hold one token per Api request which could lead quickly
216/// to many thousands of tokens.
217///
218/// Keycloak can hold multiple tokens for the same client, so we do not care
219/// about using the previous token if one thread read it and while didn´t send
220/// it yet other thread wrote it. As long as it is not expired, we can reuse it.
221/// The same goes for scalability as each container can hold a different
222/// token being all valid.
223#[derive(Debug, Default)]
224pub struct LastDatafixAccessToken(RwLock<Option<TokenResponseExtended>>);
225
226impl LastDatafixAccessToken {
227    /// Initialize a new `LastDatafixAccessToken`.
228    #[must_use]
229    pub const fn init() -> Self {
230        LastDatafixAccessToken(RwLock::new(None))
231    }
232}

234/// Reads the access token if it has been requested successfully before and it
235/// is not expired.
236#[instrument(skip(lst_acc_tkn))]
237#[allow(clippy::cast_sign_loss)]
238async fn read_access_token(
239    client_id: &str,
240    client_secret: &str,
241    tenant_id: &str,
242    lst_acc_tkn: &LastDatafixAccessToken,
243) -> Option<PubKeycloakAdminToken> {
244    let token_resp_ext_opt = match lst_acc_tkn.0.read() {
245        Ok(read) => read.clone(),
246        Err(err) => {
247            warn!("Error acquiring read lock {err:?}");
248            return None;
249        }
250    };
251
252    if let Some(data) = token_resp_ext_opt {
253        #[allow(clippy::arithmetic_side_effects, clippy::cast_possible_wrap)]
254        let pre_expiration_time: i64 =
255            data.token_resp.expires_in as i64 - PRE_EXPIRATION_SECS; // Renew the token 5 seconds before it expires
256
257        if data.client_id.eq(client_id)
258            && data.client_secret.eq(client_secret)
259            && data.tenant_id.eq(tenant_id)
260            && pre_expiration_time.is_positive()
261            && data.stamp.elapsed()
262                < Duration::from_secs({ pre_expiration_time as u64 })
263        {
264            return Some(data.token_resp);
265        }
266    }
267    return None;
268}

270/// Request a new access token and writes it to the cache
271#[instrument(err, skip(lst_acc_tkn))]
272async fn request_access_token(
273    client_id: String,
274    client_secret: String,
275    tenant_id: String,
276    lst_acc_tkn: &LastDatafixAccessToken,
277) -> AnyhowResult<PubKeycloakAdminToken> {
278    let stamp: Instant = Instant::now(); // Capture the stamp before sending the request
279    info!("Requesting access token");
280    let keycloak_adm_tkn = get_third_party_client_access_token(
281        client_id.clone(),
282        client_secret.clone(),
283        tenant_id.clone(),
284    )
285    .await?;
286
287    let token_resp: PubKeycloakAdminToken = keycloak_adm_tkn.try_into()?;
288    let mut write = match lst_acc_tkn.0.write() {
289        Ok(write) => write,
290        Err(err) => {
291            return Err(anyhow!("Error acquiring write lock: {err:?}"));
292        }
293    };
294    *write = Some(TokenResponseExtended {
295        token_resp: token_resp.clone(),
296        stamp,
297        client_id,
298        client_secret,
299        tenant_id,
300    });
301
302    Ok(token_resp)
303} // release the lock

305#[rocket::async_trait]
306impl<'r> FromRequest<'r> for DatafixClaims {
307    type Error = ();
308    #[instrument(skip_all)]
309    async fn from_request(
310        request: &'r Request<'_>,
311    ) -> Outcome<Self, Self::Error> {
312        let (tenant_id, datafix_event_id, authorization) =
313            if let Some(datafix_headers) =
314                parse_datafix_headers(request.headers())
315            {
316                (
317                    datafix_headers.tenant_id,
318                    datafix_headers.event_id,
319                    datafix_headers.authorization,
320                )
321            } else {
322                error!("DatafixClaims guard: Missing headers!");
323                return Outcome::Error((Status::BadRequest, ()));
324            };
325
326        // Try to read the access token from the cache, if it´s not there or
327        // expired request a new one and write it to the cache.
328        let lst_acc_tkn = try_outcome!(
329            request.guard::<&State<LastDatafixAccessToken>>().await
330        );
331        let token_resp = match read_access_token(
332            &authorization.client_id,
333            &authorization.client_secret,
334            &tenant_id,
335            lst_acc_tkn,
336        )
337        .await
338        {
339            Some(token_resp) => token_resp,
340            None => {
341                match request_access_token(
342                    authorization.client_id,
343                    authorization.client_secret,
344                    tenant_id.clone(),
345                    lst_acc_tkn,
346                )
347                .await
348                {
349                    Ok(token_resp) => token_resp,
350                    Err(err) => {
351                        error!("DatafixClaims guard: request_access_token error {err:?}");
352                        return Outcome::Error((Status::Unauthorized, ()));
353                    }
354                }
355            }
356        };
357
358        match decode_jwt(&token_resp.access_token) {
359            Ok(jwt_claims) => Outcome::Success(DatafixClaims {
360                jwt_claims,
361                tenant_id,
362                datafix_event_id,
363            }),
364            Err(err) => {
365                warn!("DatafixClaims guard: decode_jwt error {err:?}");
366                Outcome::Error((Status::Unauthorized, ()))
367            }
368        }
369    }
370}
