16/// Default language code to fall back to if no translation is found for a given language.
17pub const DEFAULT_LANG: &str = "en";

19/// Parses a presentation `Value` into type P.
20///
21/// Returns None if the presentation is None or deserialization fails.
22fn parse_presentation<P>(presentation: Option<&serde_json::Value>) -> Option<P>
23where
24    P: for<'de> serde::Deserialize<'de>,
25{
26    let val = presentation?;
27    deserialize_value::<P>(val.clone()).ok()
28}

30/// Generic i18n getter for nested shape: `I18nContent<I18nContent<Option<String>>>`.
31/// Reads `field` in this order: `language` -> `DEFAULT_LANG`.
32fn i18n_field(
33    i18n: Option<&I18nContent<I18nContent<Option<String>>>>,
34    language: &str,
35    field: &'static str,
36) -> Option<String> {
37    let i18n = i18n.as_ref()?;
38
39    // Try requested language first, then default language.
40    i18n.get(language)
41        .and_then(|m| m.get(field))
42        .cloned()
43        .flatten()
44        .or_else(|| {
45            i18n.get(DEFAULT_LANG)
46                .and_then(|m| m.get(field))
47                .cloned()
48                .flatten()
49        })
50}

66impl ElectionEvent {

68    /// Get the default language from presentation for the election event,
69    ///  falling back to "en" if not specified.
70    pub fn get_default_language(&self) -> String {
71        parse_presentation::<ElectionEventPresentation>(
72            self.presentation.as_ref(),
73        )
74        .and_then(|p| p.language_conf)
75        .and_then(|lc| lc.default_language_code)
76        .unwrap_or_else(|| DEFAULT_LANG.into())
77    }

158impl Alias for Election {
159    fn get_alias(&self, language: &str) -> Option<String> {
160        let base = Some(self.get_name(language));
161
162        parse_presentation::<ElectionPresentation>(self.presentation.as_ref())
163            .and_then(|p| i18n_field(p.i18n.as_ref(), language, "alias"))
164            .or(base)
165    }

168/* ===================== Contest ===================== */

170impl Name for Contest {
171    fn get_name(&self, language: &str) -> String {
172        let alias = self.alias_i18n.clone().and_then(|alias_i18n| {
173            alias_i18n
174                .get(language)
175                .cloned()
176                .or(alias_i18n.get(DEFAULT_LANG).cloned())
177                .or(Some(self.alias.clone()))
178                .flatten()
179        });
180        let name = self.name_i18n.clone().and_then(|name_i18n| {
181            name_i18n
182                .get(language)
183                .cloned()
184                .or(name_i18n.get(DEFAULT_LANG).cloned())
185                .or(Some(self.name.clone()))
186                .flatten()
187        });
188
189        alias.or(name).unwrap_or("-".into())
190    }
191}
