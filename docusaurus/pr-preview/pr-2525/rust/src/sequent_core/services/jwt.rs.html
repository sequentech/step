76#[instrument(err, skip_all)]
77/// Decodes a JWT token string into `JwtClaims`.
78///
79/// # Errors
80/// Returns an error if the token is malformed, cannot be base64 decoded, is not valid UTF-8, or cannot be deserialized into `JwtClaims`.
81#[allow(clippy::arithmetic_side_effects)]
82pub fn decode_jwt(token: &str) -> Result<JwtClaims> {
83    let parts: Vec<&str> = token.split('.').collect();
84    let part = parts.get(1).ok_or(anyhow::anyhow!("Bad token (no '.')"))?;
85    let bytes = general_purpose::URL_SAFE_NO_PAD
86        .decode(part)
87        .map_err(|err| anyhow!("Error decoding string: {err:?}"))?;
88    let json = String::from_utf8(bytes)
89        .map_err(|err| anyhow!("Error decoding bytes to utf8: {err:?}"))?;
90    debug!("json: {json:?}");
91    let claims: JwtClaims = serde_json::from_str(&json).map_err(|err| {
92        anyhow!("Error decoding string into formatted json: {err:?}")
93    })?;
94
95    Ok(claims)
96}

98/// Decodes the permission labels from the JWT claims.
99///
100/// # Errors
101/// Returns an empty vector if no permission labels are present.
102#[instrument(skip_all, ret)]
103#[allow(clippy::arithmetic_side_effects)]
104pub fn decode_permission_labels(claims: &JwtClaims) -> Vec<String> {
105    let Some(label_str) = claims.hasura_claims.permission_labels.clone() else {
106        return vec![];
107    };
108
109    let s = label_str.trim();
110    let s = if s.starts_with('{') && s.ends_with('}') {
111        &s[1..s.len() - 1]
112    } else {
113        s
114    };
115
116    // Split the string into items
117    let items = s.split(',');
118
119    // Process each item: trim whitespace and surrounding quotes
120    let keys: Vec<String> = items
121        .map(|item| item.trim().trim_matches('"').to_string())
122        .filter(|item| !item.is_empty())
123        .collect();
124    keys
125}

127/**
128 * Returns true only if the JWT has gold permissions and the JWT
129 * authentication is fresh, i.e. performed less than 60 seconds ago.
130 */
131#[instrument(skip_all)]
132#[allow(clippy::arithmetic_side_effects)]
133pub fn has_gold_permission(claims: &JwtClaims) -> bool {
134    let auth_time_local: DateTime<Local> =
135        if let Some(auth_time_int) = claims.auth_time {
136            if let Ok(auth_time_parsed) =
137                ISO8601::timestamp_ms_utc_to_date_opt(auth_time_int * 1000)
138            {
139                auth_time_parsed
140            } else {
141                warn!("Failed to parse auth_time");
142                return false;
143            }
144        } else {
145            info!("fallback to iat, claims.auth_time is None");
146            if let Ok(iat_parsed) =
147                ISO8601::timestamp_ms_utc_to_date_opt(claims.iat * 1000)
148            {
149                iat_parsed
150            } else {
151                warn!("Failed to parse iat");
152                return false;
153            }
154        };
155    // Let's asume fresh means token has at most 1 minute since authentication
156    let freshness_limit = ISO8601::now() - Duration::seconds(60);
157    let is_fresh = auth_time_local > freshness_limit;
158    warn!("is_fresh={is_fresh:?}, auth_time_local={auth_time_local:?}, freshness_limit={freshness_limit:?}");
159    let is_gold = claims.acr == Permissions::GOLD.to_string();
160    is_fresh && is_gold
161}
