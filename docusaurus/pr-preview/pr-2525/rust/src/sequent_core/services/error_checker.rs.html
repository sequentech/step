7use crate::{

10        DecodedVoteContest, InvalidPlaintextError, InvalidPlaintextErrorType,
11    },

14/// Checks the maximum selections per candidate type in a contest.
15///
16/// # Panics
17/// Panics if the count overflows when incrementing selections per type.
18#[must_use]
19pub fn check_max_selections_per_type(
20    contest: &Contest,
21    decoded_vote: &DecodedVoteContest,
22) -> Vec<InvalidPlaintextError> {
23    let presentation = contest.presentation.clone().unwrap_or_default();
24    let Some(max_selections_per_type) = presentation.max_selections_per_type
25    else {
26        return vec![];
27    };
28
29    let mut candidates_type_map = HashMap::<String, String>::new();
30
31    for candidate in &contest.candidates {
32        if let Some(candidate_type) = candidate.candidate_type.clone() {
33            candidates_type_map.insert(candidate.id.clone(), candidate_type);
34        }
35    }
36
37    let mut type_count = HashMap::<String, u64>::new();
38
39    for selection in &decoded_vote.choices {
40        if selection.selected < 0 {
41            continue;
42        }
43        let Some(candidate_type) = candidates_type_map.get(&selection.id)
44        else {
45            continue;
46        };
47        let current_count = type_count.get(candidate_type).unwrap_or(&0);
48        let count_to_insert = current_count
49            .checked_add(1)
50            .expect("Overflow when counting selections per type");
51        type_count.insert(candidate_type.clone(), count_to_insert);
52    }
53
54    let mut invalid_errors = vec![];
55
56    for (key, value) in &type_count {
57        if *value > max_selections_per_type {
58            invalid_errors.push(InvalidPlaintextError {
59                error_type: InvalidPlaintextErrorType::Implicit,
60                candidate_id: None,
61                message: Some(
62                    "errors.implicit.maxSelectionsPerType".to_string(),
63                ),
64                message_map: HashMap::from([
65                    ("type".to_string(), key.clone()),
66                    ("numSelected".to_string(), value.to_string()),
67                    ("max".to_string(), max_selections_per_type.to_string()),
68                ]),
69            });
70        }
71    }
72
73    invalid_errors
74}
75
76#[must_use]
77pub fn check_contest(
78    contest: &Contest,
79    decoded_vote: &DecodedVoteContest,
80) -> DecodedVoteContest {
81    let mut with_errors = decoded_vote.clone();
82
83    let mut invalid_errors = decoded_vote.invalid_errors.clone();
84
85    // check max selections per type
86    invalid_errors.extend(check_max_selections_per_type(contest, decoded_vote));
87
88    with_errors.invalid_errors = invalid_errors;
89    with_errors
90}
