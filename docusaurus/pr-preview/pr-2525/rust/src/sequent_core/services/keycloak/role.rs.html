11impl From<GroupRepresentation> for Role {

41impl KeycloakAdminClient {
42    /// Lists roles in the given realm.
43    ///
44    /// # Errors
45    /// Returns an error if the request fails or the API call is unsuccessful.
46    #[instrument(skip(self), err)]
47    pub async fn list_roles(
48        self,
49        realm: &str,
50        search: Option<String>,
51        limit: Option<usize>,
52        offset: Option<usize>,
53    ) -> Result<(Vec<Role>, usize)> {
54        let group_representations: Vec<GroupRepresentation> = self
55            .client
56            .realm_groups_get(
57                realm.clone(),
58                Some(false),
59                None,
60                None,
61                None,
62                None,
63                search.clone(),
64                None,
65            )
66            .await
67            .map_err(|err| anyhow!("{err:?}"))?;
68
69        let count = group_representations.len();
70        let start = offset.unwrap_or(0);
71        let end = match limit {
72            Some(num) => count.min(start.saturating_add(num)),
73            None => count,
74        };
75        let slized_group_representations =
76            group_representations.get(start..end).ok_or_else(|| {
77                anyhow!("Invalid slice range for group representations")
78            })?;
79        let roles = slized_group_representations
80            .iter()
81            .map(|role| role.clone().into())
82            .collect();
83        Ok((roles, count))
84    }

86    /// Lists user roles in the given realm.
87    ///
88    /// # Errors
89    /// Returns an error if the request fails or the API call is unsuccessful.
90    #[instrument(skip(self), err)]
91    pub async fn list_user_roles(
92        self: &KeycloakAdminClient,
93        realm: &str,
94        user_id: &str,
95    ) -> Result<Vec<Role>> {
96        let groups: Vec<GroupRepresentation> = self
97            .client
98            .realm_users_with_user_id_groups_get(
99                realm,
100                user_id,
101                Some(false),
102                None,
103                None,
104                None,
105            )
106            .await
107            .map_err(|err| anyhow!("{err:?}"))?;
108        let roles = groups.into_iter().map(std::convert::Into::into).collect();
109        Ok(roles)
110    }

165    /// Creates a role in the given realm.
166    ///
167    /// # Errors
168    /// Returns an error if the request fails or the API call is unsuccessful.
169    #[instrument(skip(self), err)]
170    pub async fn create_role(self, realm: &str, role: &Role) -> Result<Role> {
171        self.client

173            .await
174            .map_err(|err| anyhow!("{err:?}"))?;
175        Ok(role.clone())
176    }

178    /// Gets a role by name in the given realm.
179    ///
180    /// # Errors
181    /// Returns an error if the request fails or the API call is unsuccessful.
182    #[instrument(skip(self), err)]
183    pub async fn get_role_by_name(
184        self,
185        realm: &str,
186        role: &Role,
187    ) -> Result<Role> {
188        let (roles, count) = self.list_roles(realm, None, None, None).await?;
189        let role_by_named = roles.iter().find(|r| role.name == r.name);
190        let new_role = match role_by_named {
191            Some(new_rolee) => new_rolee,
192            None => role,
193        };
194        Ok(new_role.clone())
195    }
