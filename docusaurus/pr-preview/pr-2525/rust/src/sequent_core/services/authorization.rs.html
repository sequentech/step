16/// Authorizes a user based on JWT claims, tenant, and permissions.
17///
18/// # Errors
19/// Returns an error if the user is not authorized or required environment variables are missing.
20pub fn authorize(
21    claims: &JwtClaims,
22    allow_super_admin_auth: bool, // Allow authorizing super admin tenant
23    tenant_id_opt: Option<String>,
24    permissions: Vec<Permissions>,
25) -> Result<(), (Status, String)> {
26    // Verify tenant id
27    let allowed = match (tenant_id_opt.clone(), allow_super_admin_auth) {
28        (Some(tenant_id), _)
29            if tenant_id.eq(&claims.hasura_claims.tenant_id) =>
30        {
31            true // is valid tenant
32        }
33
34        (_, true) => {
35            let super_admin_tenant_id = env::var("SUPER_ADMIN_TENANT_ID")
36                .map_err(|_| {
37                    (
38                        Status::Unauthorized,
39                        "SUPER_ADMIN_TENANT_ID must be set".to_string(),
40                    )
41                })?;
42            info!("super_admin_tenant_id: {super_admin_tenant_id}");
43            super_admin_tenant_id == claims.hasura_claims.tenant_id // is super admin?
44        }
45        (_, _) => false, // Is not valid tenant nor super admin
46    };
47
48    if !allowed {
49        error!(
50            "Not authorized: allow_super_admin_auth: {allow_super_admin_auth}, 
51            tenant_id_opt: {tenant_id_opt:?}, claims tenant_id: {}",
52            claims.hasura_claims.tenant_id
53        );
54        return Err((Status::Unauthorized, format!("Unauthorized: not a super admin or invalid tenant_id {tenant_id_opt:?}")));
55    }
56
57    let perms_str: Vec<String> = permissions
58        .into_iter()
59        .map(|permission| permission.to_string())
60        .collect();
61    let permissions_set: HashSet<_> =
62        claims.hasura_claims.allowed_roles.iter().collect();
63    let all_contained =
64        perms_str.iter().all(|item| permissions_set.contains(&item));
65
66    if all_contained {
67        Ok(())
68    } else {
69        Err((
70            Status::Unauthorized,
71            format!("Unathorized: {perms_str:?} not in {permissions_set:?}"),
72        ))
73    }
74}

77/// Authorizes a voter for an election based on JWT claims and permissions.
78// / Returns area_id
79///
80/// # Errors
81/// Returns an error if the voter is not authorized for the election or required claims are missing.
82pub fn authorize_voter_election(
83    claims: &JwtClaims,
84    permissions: Vec<VoterPermissions>,
85    election_id: &String,
86) -> Result<(String, VotingStatusChannel), (Status, String)> {
87    let perms_str: Vec<String> = permissions
88        .into_iter()
89        .map(|permission| permission.to_string())
90        .collect();
91    let permissions_set: HashSet<_> =
92        claims.hasura_claims.allowed_roles.iter().collect();
93    let all_contained =
94        perms_str.iter().all(|item| permissions_set.contains(&item));
95
96    if !all_contained {
97        return Err((Status::Unauthorized, String::new()));
98    }
99
100    let Some(area_id) = claims.hasura_claims.area_id.clone() else {
101        return Err((Status::Unauthorized, "Missing area_id".into()));
102    };
103
104    // Check election id checks
105    if claims.hasura_claims.authorized_election_ids.is_none()
106        || !claims
107            .hasura_claims
108            .authorized_election_ids
109            .as_ref()
110            .unwrap_or(&Vec::new())
111            .contains(election_id)
112    {
113        return Err((
114            Status::Unauthorized,
115            "Not authorized to election".into(),
116        ));
117    }
118
119    match claims.azp.as_str() {
120        "voting-portal" => Ok((area_id, VotingStatusChannel::ONLINE)),
121        "voting-portal-kiosk" => Ok((area_id, VotingStatusChannel::KIOSK)),
122        _ => Err((Status::Unauthorized, "Unknown Client".into())),
123    }
124}
