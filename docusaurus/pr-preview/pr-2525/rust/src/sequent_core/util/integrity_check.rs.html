4use std::fmt::Write;
5use std::fs::File;
6use std::io::Read;
7use strand::hash::hash_sha256;
8use strand::util::StrandError;
9use strum_macros::Display;
10use tempfile::NamedTempFile;
11
12#[derive(Debug, Display)]
13/// Errors that can occur during the integrity check of a file.
14pub enum HashFileVerifyError {
15    #[strum(serialize = "io-error")]
16    /// Error reading voters file
17    IoError(String, std::io::Error),
18    #[strum(serialize = "hash-mismatch")]
19    /// Voters file hash does not match
20    HashMismatch(String, String),
21    #[strum(serialize = "hash-computing-error")]
22    /// Error computing the hash
23    HashComputingError(String, StrandError),
24}
25
26impl std::error::Error for HashFileVerifyError {}
27
28/// Checks the integrity of a file by comparing its SHA-256 hash.
29///
30/// # Errors
31/// Returns an error if the file cannot be opened, read, or if the hash does not match.
32pub fn integrity_check(
33    temp_file_path: &NamedTempFile,
34    sha256: String,
35) -> Result<(), HashFileVerifyError> {
36    let mut sha256 = sha256;
37    sha256.make_ascii_lowercase();
38    let mut file = File::open(temp_file_path).map_err(|err| {
39        HashFileVerifyError::IoError(
40            "Error opening the temp file.".to_string(),
41            err,
42        )
43    })?;
44
45    let mut file_buffer: Vec<u8> = vec![];
46    file.read_to_end(&mut file_buffer).map_err(|err| {
47        HashFileVerifyError::IoError(
48            "Error reading the temp file.".to_string(),
49            err,
50        )
51    })?;
52
53    let calculated_hash_result = hash_sha256(file_buffer.as_slice());
54    match calculated_hash_result {
55        Ok(hash) => {
56            // Get lowercase hex representation.
57            let hash_str: String =
58                hash.iter().fold(String::new(), |mut output, b| {
59                    let _ = write!(output, "{b:02x}");
60                    output
61                });
62            if !hash_str.eq(sha256.as_str()) {
63                return Err(HashFileVerifyError::HashMismatch(
64                    sha256, hash_str,
65                ));
66            }
67        }
68        Err(err) => {
69            return Err(HashFileVerifyError::HashComputingError(
70                "Error computing the hash from file.".to_string(),
71                err,
72            ));
73        }
74    }
75    Ok(())
76}
