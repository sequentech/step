200#[must_use]
201/// Normalizes an external locale representation into the internal language code
202/// used by the web frontends.
203///
204/// Inputs may be BCP 47 tags (for example `ca` or `es-ES`), already-internal
205/// language codes (for example `cat`), or ISO 639-2/T codes (for example
206/// `eng`). The returned value is the internal application language code.
207pub fn locale_to_internal_language_code(lang: &str) -> String {
208    let primary = lang.split('-').next().unwrap_or(lang).to_ascii_lowercase();
209    match iso_639_2t_to_bcp47(primary.as_str()) {
210        "ca" => "cat".to_string(),
211        normalized => normalized.to_string(),
212    }
213}

215#[cfg(test)]
216mod tests {
217    use super::*;
218
219    #[test]
220    fn test_iso_639_2t_catalan() {
221        assert_eq!(iso_639_2t_to_bcp47("cat"), "ca");
222    }
223
224    #[test]
225    fn test_iso_639_2t_already_bcp47() {
226        assert_eq!(iso_639_2t_to_bcp47("en"), "en");
227        assert_eq!(iso_639_2t_to_bcp47("es"), "es");
228        assert_eq!(iso_639_2t_to_bcp47("fr"), "fr");
229        assert_eq!(iso_639_2t_to_bcp47("nl"), "nl");
230        assert_eq!(iso_639_2t_to_bcp47("eu"), "eu");
231        assert_eq!(iso_639_2t_to_bcp47("gl"), "gl");
232        assert_eq!(iso_639_2t_to_bcp47("tl"), "tl");
233    }
234
235    #[test]
236    fn test_iso_639_2t_common_languages() {
237        assert_eq!(iso_639_2t_to_bcp47("fra"), "fr");
238        assert_eq!(iso_639_2t_to_bcp47("deu"), "de");
239        assert_eq!(iso_639_2t_to_bcp47("spa"), "es");
240        assert_eq!(iso_639_2t_to_bcp47("por"), "pt");
241        assert_eq!(iso_639_2t_to_bcp47("rus"), "ru");
242        assert_eq!(iso_639_2t_to_bcp47("ara"), "ar");
243        assert_eq!(iso_639_2t_to_bcp47("zho"), "zh");
244        assert_eq!(iso_639_2t_to_bcp47("jpn"), "ja");
245    }
246
247    #[test]
248    fn test_iso_639_2t_unknown_passthrough() {
249        assert_eq!(iso_639_2t_to_bcp47("xyz"), "xyz");
250    }
251
252    #[test]
253    fn test_locale_to_internal_language_code_catalan() {
254        assert_eq!(locale_to_internal_language_code("ca"), "cat");
255        assert_eq!(locale_to_internal_language_code("ca-ES"), "cat");
256        assert_eq!(locale_to_internal_language_code("cat"), "cat");
257    }
258
259    #[test]
260    fn test_locale_to_internal_language_code_common_languages() {
261        assert_eq!(locale_to_internal_language_code("en"), "en");
262        assert_eq!(locale_to_internal_language_code("en-US"), "en");
263        assert_eq!(locale_to_internal_language_code("eng"), "en");
264        assert_eq!(locale_to_internal_language_code("es-ES"), "es");
265        assert_eq!(locale_to_internal_language_code("spa"), "es");
266    }
