5use anyhow::{Context, Result};
6use chrono::{DateTime, Duration, Local, Utc};
7use std::time::{SystemTime, UNIX_EPOCH};
8
9/// Get the current system date in the format "day/month/year".
10#[must_use]
11pub fn get_current_date() -> String {
12    let local: DateTime<Local> = Local::now();
13    local.format("%-d/%-m/%Y").to_string()
14}
15
16/// Get the timestamp for a given number of seconds later than the current time.
17/// # Panics
18/// If the addition of seconds results in an overflow.
19#[must_use]
20pub fn get_seconds_later(seconds: i64) -> DateTime<Utc> {
21    let current_time = Utc::now();
22    current_time
23        .checked_add_signed(Duration::seconds(seconds))
24        .expect("Overflow when adding seconds to current time")
25}
26
27/// Get the current timestamp in seconds since the UNIX epoch.
28/// # Errors
29/// If the current system time is before the `UNIX_EPOCH`, which is highly unlikely.
30pub fn timestamp() -> Result<u64> {
31    Ok(SystemTime::now()
32        .duration_since(UNIX_EPOCH)
33        .with_context(|| "Impossible with respect to UNIX_EPOCH")?
34        .as_secs())
35}
