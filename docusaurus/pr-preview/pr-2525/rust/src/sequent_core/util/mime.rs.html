5/// Returns all MIME types for the given file extension.
6/// Falls back to `["application/json"]` if none is found.
7#[must_use]
8pub fn get_mime_types(extension: &str) -> &'static [&'static str] {
9    // Static array of (extension, list-of-mime-types) tuples.
10    static MIME_TYPES: [(&str, &[&str]); 27] = [
11        ("jpg",  &["image/jpeg"]),
12        ("jpeg", &["image/jpeg"]),
13        ("png",  &["image/png"]),
14        ("gif",  &["image/gif"]),
15        ("bmp",  &["image/bmp"]),
16        ("webp", &["image/webp"]),
17        ("svg",  &["image/svg+xml"]),
18        ("txt",  &["text/plain"]),
19        ("html", &["text/html"]),
20        ("htm",  &["text/html"]),
21        ("csv",  &["text/csv"]),
22        ("json", &["application/json"]),
23        ("pdf",  &["application/pdf"]),
24        ("doc",  &["application/msword"]),
25        (
26            "docx",
27            &["application/vnd.openxmlformats-officedocument.wordprocessingml.document"],
28        ),
29        ("xls",  &["application/vnd.ms-excel"]),
30        (
31            "xlsx",
32            &["application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"],
33        ),
34        ("ppt",  &["application/vnd.ms-powerpoint"]),
35        (
36            "pptx",
37            &["application/vnd.openxmlformats-officedocument.presentationml.presentation"],
38        ),
39        ("mp3",  &["audio/mpeg"]),
40        ("wav",  &["audio/wav"]),
41        ("mp4",  &["video/mp4"]),
42        ("avi",  &["video/x-msvideo"]),
43        ("mov",  &["video/quicktime"]),
44        // For ZIP, multiple known MIME types:
45        ("zip",  &["application/zip", "application/x-zip-compressed", "multipart/x-zip"]),
46        ("tar",  &["application/x-tar"]),
47        ("gz",   &["application/gzip"]),
48    ];
49
50    // Simple linear lookup through our static array
51    for (ext, mimes) in &MIME_TYPES {
52        if *ext == extension {
53            return mimes;
54        }
55    }
56
57    // Fallback MIME if none found
58    &["application/json"]
59}
60
61/// Checks if a given extension is associated with the specified MIME type.
62#[must_use]
63pub fn matches_mime(extension: &str, mime_type: &str) -> bool {
64    get_mime_types(extension).contains(&mime_type)
65}

67#[cfg(test)]
68mod tests {
69    use super::*;
70
71    #[test]
72    fn test_zip_mime_types() {
73        let zip_mimes = get_mime_types("zip");
74        assert_eq!(zip_mimes.len(), 3);
75        assert!(zip_mimes.contains(&"application/zip"));
76        assert!(zip_mimes.contains(&"application/x-zip-compressed"));
77        assert!(zip_mimes.contains(&"multipart/x-zip"));
78    }
79
80    #[test]
81    fn test_matches_mime() {
82        assert!(matches_mime("zip", "application/zip"));
83        assert!(!matches_mime("zip", "application/pdf"));
84    }
85
86    #[test]
87    fn test_unknown_extension() {
88        // Unknown extension should return the default "application/json"
89        let mimes = get_mime_types("unknownext");
90        assert_eq!(mimes, ["application/json"]);
91    }
92}
