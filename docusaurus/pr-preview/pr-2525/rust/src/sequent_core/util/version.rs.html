10pub const ENV_VAR_APP_VERSION: &str = "APP_VERSION";
11pub const ENV_VAR_APP_HASH: &str = "APP_HASH";
12pub const VERSION_KEY: &str = "version";
13pub const HISTORICAL_DEFAULT_VERSION: &str = "9.0.0";

15/// Checks if the imported version is compatible with the current version.
16///
17/// # Errors
18/// Returns an error if the imported version is not compatible with the current version.
19pub fn check_version_compatibility(
20    imported_version: &str,
21    current_version: &str,
22) -> Result<()> {
23    #[cfg(feature = "log")]
24    info!(
25        "Checking version compatibility - Current: {}, Imported: {}",
26        current_version, imported_version
27    );
28
29    // If current version is DEV_APP_VERSION, allow any import
30    if current_version == DEV_APP_VERSION {
31        #[cfg(feature = "log")]
32        info!("Current version is 'dev', allowing import");
33        return Ok(());
34    }
35
36    if imported_version == DEV_APP_VERSION {
37        #[cfg(feature = "log")]
38        info!("Imported version is 'dev' while system is not in dev mode, rejecting import");
39        return Err(anyhow!("Imported version is 'dev', which is not compatible with current version {current_version}. Please use a different version."));
40    }
41
42    let (current_major, current_minor, _) = extract_semver(current_version)
43        .ok_or_else(|| anyhow!("Could not parse current version"))?;
44    let (imported_major, imported_minor, _) = extract_semver(imported_version)
45        .ok_or_else(|| anyhow!("Could not parse imported version"))?;
46
47    if (current_major != imported_major) || (current_minor < imported_minor) {
48        return Err(anyhow!(
49            "Version mismatch: Imported version {imported_version} is not compatible with current version {current_version}. Please upgrade your system."
50        ));
51    }
52    Ok(())
53}

55/// Extract the semantic version from the given string.
56///
57/// Major version is required, and None will be returned if missing.
58/// Missing minor or patch versions will default to 0; pre-release tags are ignored (-*).
59fn extract_semver(input: &str) -> Option<(u64, u64, u64)> {
60    // Trim optional 'v' or 'V' prefix
61    let trimmed = input.trim_start_matches(['v', 'V']);
62
63    // Split off any pre-release suffix (e.g. "-alpha", "-rc1") by splitting on '-'
64    let version_part = trimmed.split('-').next().unwrap_or("");
65
66    // Split on '.' and parse up to three parts; missing components default to 0
67    let mut parts = version_part.split('.').map(|s| s.parse().ok());
68    let major: u64 = parts.next()??;
69    let minor: u64 = parts.next().flatten().unwrap_or(0);
70    let patch: u64 = parts.next().flatten().unwrap_or(0);
71
72    Some((major, minor, patch))
73}

83    #[test]
84    fn test_current_version_is_dev() {
85        // If current system is DEV_APP_VERSION, it should accept anything
86        assert!(check_version_compatibility("1.0.0", DEV_APP_VERSION).is_ok());
87        assert!(
88            check_version_compatibility("99.99.99", DEV_APP_VERSION).is_ok()
89        );
90        assert!(
91            check_version_compatibility(DEV_APP_VERSION, DEV_APP_VERSION)
92                .is_ok()
93        );
94    }
95
96    #[test]
97    fn test_imported_version_is_dev_rejected() {
98        // If importing DEV_APP_VERSION into a non-dev system, it must fail
99        let result = check_version_compatibility(DEV_APP_VERSION, "1.0.0");
100        assert!(result.is_err());
101        assert_eq!(
102            result.unwrap_err().to_string(),
103            "Imported version is 'dev', which is not compatible with current version 1.0.0. Please use a different version."
104        );
105    }
106
107    #[test]
108    fn test_exact_match_versions() {
109        assert!(check_version_compatibility("1.0.0", "1.0.0").is_ok());
110        assert!(check_version_compatibility("2.5.1", "2.5.1").is_ok());
111    }
112
113    #[test]
114    fn test_backward_compatibility() {
115        // Importing an OLDER version into a NEWER system should be OK
116        // Imported: 1.0.0, Current: 1.1.0
117        assert!(check_version_compatibility("1.0.0", "1.1.0").is_ok());
118
119        // Imported: 10, Current: 11
120        assert!(check_version_compatibility("10.5.5", "10.5.7").is_ok());
121    }
122
123    #[test]
124    fn test_forward_compatibility_rejection() {
125        // Importing a NEWER version into an OLDER system should FAIL
126        // Imported: 1.1.0, Current: 1.0.0
127        let result = check_version_compatibility("1.1.0", "1.0.0");
128        assert!(result.is_err());
129        assert!(result.unwrap_err().to_string().contains("not compatible"));
130
131        // Patch version is accepted.
132        let result = check_version_compatibility("1.1.1", "1.1.0");
133        assert!(result.is_ok());
134    }

136    #[test]
137    fn test_major_mismatch_rejection() {
138        // Importing anything with a different major version should fail,
139        //  be it older or newer.
140        let result = check_version_compatibility("1.0.0", "2.0.0");
141        assert!(result.is_err());
142        assert!(result.unwrap_err().to_string().contains("not compatible"));
143
144        let result = check_version_compatibility("11.0.0", "10.0.0");
145        assert!(result.is_err());
146        assert!(result.unwrap_err().to_string().contains("not compatible"));
147    }

149    #[test]
150    fn test_parsing_failures() {
151        // Invalid current version
152        let res_current = check_version_compatibility("1.0.0", "invalid_ver");
153        assert!(res_current.is_err());
154        assert!(res_current
155            .unwrap_err()
156            .to_string()

158
159        // Invalid imported version

161        assert!(res_imported.is_err());
162        assert!(res_imported

164            .to_string()

171        // Imported v1 (1) into Current 1 -> OK

174        // Imported V2 (2) into Current v1 (1) -> Error

182    #[test]
183    fn test_extract_semver_logic() {
184        // Standard semver
185        assert_eq!(extract_semver("1.2.3"), Some((1, 2, 3)));
186        assert_eq!(extract_semver("10.0.0"), Some((10, 0, 0)));
187        assert_eq!(extract_semver("0.5.9"), Some((0, 5, 9)));
188
189        // With prefixes
190        assert_eq!(extract_semver("v1.2.3"), Some((1, 2, 3)));
191        assert_eq!(extract_semver("V2.0.0"), Some((2, 0, 0)));
192
193        // With suffixes (alpha, beta, rc)
194        assert_eq!(extract_semver("1.0.0-alpha"), Some((1, 0, 0)));
195        assert_eq!(extract_semver("3.0.0-rc1"), Some((3, 0, 0)));
196        assert_eq!(extract_semver("v4-beta"), Some((4, 0, 0)));
197
198        // Edge cases
199        assert_eq!(extract_semver("2"), Some((2, 0, 0))); // Just a number
200        assert_eq!(extract_semver("2.none.1"), Some((2, 0, 1))); // Parse error in the midle
201        assert_eq!(extract_semver("q.1.0"), None); // Major version is problematic
202        assert_eq!(extract_semver("not_a_number"), None);
203        assert_eq!(extract_semver(""), None);
204        assert_eq!(extract_semver("v"), None);
205
206        // Ensure it stops at non-digits
207        assert_eq!(extract_semver("5startswithnumber"), None);
208    }
