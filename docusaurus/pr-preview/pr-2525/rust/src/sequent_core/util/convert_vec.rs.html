5use serde_json::Value;
6use std::{collections::HashMap, hash::BuildHasher};
7
8/// A trait for converting a value into a `Vec<String>`.
9pub trait IntoVec {
10    /// Converts a value into a `Vec<String>`.
11    fn into_vec(self) -> Vec<String>;
12}
13
14impl IntoVec for String {
15    fn into_vec(self) -> Vec<String> {
16        vec![self]
17    }
18}
19
20impl IntoVec for Vec<String> {
21    fn into_vec(self) -> Vec<String> {
22        self
23    }
24}
25
26impl IntoVec for Value {
27    fn into_vec(self) -> Vec<String> {
28        match self {
29            Value::String(s) => vec![s],
30            Value::Array(arr) => arr
31                .into_iter()
32                .filter_map(|v| {
33                    if let Value::String(s) = v {
34                        Some(s)
35                    } else {
36                        None
37                    }
38                })
39                .collect(),
40            _ => vec![],
41        }
42    }
43}
44
45#[must_use]
46/// Converts a `HashMap<String, Value>` to a `HashMap<String, Vec<String>>`,
47///  where the `Value` can be either a `String` or an `Array` of `Strings`.
48pub fn convert_map<S>(
49    original_map: HashMap<String, Value, S>,
50) -> HashMap<String, Vec<String>, S>
51where
52    S: BuildHasher + Default,
53{
54    original_map
55        .into_iter()
56        .map(|(key, value)| {
57            let vec = match value {
58                Value::Array(arr) => arr
59                    .into_iter()
60                    .filter_map(|v| v.as_str().map(str::to_string))
61                    .collect(),
62                Value::String(s) => vec![s],
63                _ => Vec::new(),
64            };
65            (key, vec)
66        })
67        .collect()
68}
