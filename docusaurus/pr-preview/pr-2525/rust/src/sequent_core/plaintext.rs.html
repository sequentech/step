29/// Represents an error in the preferential order.
30pub enum PreferencialOrderErrorType {
31    /// Indicates that there are gaps in the preference order, e.g. 1,2,4 or 1,3,4.

47#[derive(Serialize, Deserialize, JsonSchema, PartialEq, Eq, Debug, Clone)]
48/// Represents a decoded vote contest.
49#[allow(missing_docs)]
50pub struct DecodedVoteContest {
51    pub contest_id: String,
52    pub is_explicit_invalid: bool,
53    pub invalid_errors: Vec<InvalidPlaintextError>,
54    pub invalid_alerts: Vec<InvalidPlaintextError>,
55    pub choices: Vec<DecodedVoteChoice>,
56}
57
58impl DecodedVoteContest {
59    /// Check if the contest is invalid, which is true if it is explicitly
60    ///  marked as invalid or if it has any invalid errors.
61    #[must_use]
62    pub const fn is_invalid(&self) -> bool {
63        self.is_explicit_invalid || !self.invalid_errors.is_empty()
64    }
65    /// Check if the contest is blank, which is true
66    /// if it is not invalid and all choices are unselected.
67    #[must_use]
68    pub fn is_blank(&self) -> bool {
69        !self.is_invalid()
70            && self
71                .choices
72                .clone()
73                .iter()
74                .all(|choice| choice.selected < 0)
75    }
76
77    /// Check the validity of the preference order.
78    ///
79    /// Note: `PreferenceOrderWithGaps` is returned as an error if there are gaps,
80    /// but this is generally not considered invalid, so the caller can
81    /// handle it depending on the policy or jurisdiction rules.
82    /// Returns Ok if the order is valid after sorting it and if it is
83    /// contiguous, e.g. 1,2,3,4 or 1,4,2,3.
84    /// Returns Err with a Vec of all errors found (may contain multiple variants).
85    ///
86    /// # Errors
87    /// Returns `Err(Vec<PreferencialOrderErrorType>)` if the order is invalid.
88    ///
89    /// # Panics
90    /// Panics if there are more than `i64::MAX` selected choices, which
91    /// would cause an overflow when converting from `usize` to `i64`.
92    pub fn validate_preferencial_order(
93        &self,
94    ) -> Result<(), Vec<PreferencialOrderErrorType>> {
95        let mut errors: Vec<PreferencialOrderErrorType> = Vec::new();
96
97        // Discard the unselected choices and sort the selected ones by their preference order
98        let choices: Vec<i64> = self
99            .choices
100            .iter()
101            .filter(|choice| choice.selected >= 0)
102            .map(|choice| choice.selected)
103            .collect();
104
105        // After removing the unselected choices we check that there are no duplicates in
106        // the preference order
107        let choices_unique_set = choices.iter().collect::<HashSet<_>>();
108        if choices.len() != choices_unique_set.len() {
109            errors.push(PreferencialOrderErrorType::DuplicatedPosition);
110        }
111
112        // Check that there are no gaps in the ordered choices
113        let mut ordered_choices = choices_unique_set
114            .into_iter()
115            .copied()
116            .collect::<Vec<i64>>();
117        ordered_choices.sort_unstable();
118        let expected_order: Vec<i64> = (0..ordered_choices.len())
119            .map(|i| i64::try_from(i).expect("failed to convert usize to i64"))
120            .collect();
121
122        if ordered_choices != expected_order {
123            errors.push(PreferencialOrderErrorType::PreferenceOrderWithGaps);
124        }
125
126        if errors.is_empty() {
127            Ok(())
128        } else {
129            Err(errors)
130        }
131    }
132}

134#[derive(Serialize, Deserialize, JsonSchema, PartialEq, Eq, Debug, Clone)]
135/// Represents a decoded vote choice.
136pub struct DecodedVoteChoice {
137    /// The candidate ID for this choice.
138    pub id: String,
139    /// The selection value for this choice, where -1 indicates not selected
140    /// and any non-negative value indicates selected and depends on the counting algorithm.
141    pub selected: i64,
142    /// The write-in text for this choice, if applicable.
143    pub write_in_text: Option<String>,
144}
145
146impl DecodedVoteChoice {
147    /// Check if the choice is selected.
148    #[must_use]
149    pub const fn is_selected(&self) -> bool {
150        self.selected >= 0
151    }
152}

154/// Maps an auditable ballot to decoded contests.
155///
156/// # Errors
157/// Returns `Err(String)` if the number of contests does not match or if deserialization fails.
158pub fn map_to_decoded_contest<C: Ctx<P = [u8; 30]>>(
159    ballot: &AuditableBallot,
160) -> Result<Vec<DecodedVoteContest>, String> {
161    let mut decoded_contests = vec![];
162    if ballot.config.contests.len() != ballot.contests.len() {
163        return Err(format!(
164            "Invalid number of contests {} != {}",
165            ballot.config.contests.len(),
166            ballot.contests.len()
167        ));
168    }
169
170    let ballot_contests = ballot.deserialize_contests().map_err(|err| {
171        format!("Error deserializing auditable ballot contest {err:?}")
172    })?;
173    for contest in &ballot_contests {
174        let found_contest = ballot
175            .config
176            .contests
177            .iter()
178            .find(|contest_el| contest_el.id == contest.contest_id)
179            .ok_or_else(|| {
180                format!(
181                    "Can't find contest with id {} on ballot style",
182                    contest.contest_id
183                )
184            })?;
185        let replication_choice: &ReplicationChoice<C> = &contest.choice;
186        let decoded_plaintext = found_contest
187            .decode_plaintext_contest(&replication_choice.plaintext)?;
188        decoded_contests.push(decoded_plaintext);
189    }
190    Ok(decoded_contests)
191}

193/// Maps decoded ballot choices to decoded contests.
194///
195/// # Errors
196/// Returns `Err(String)` if a contest cannot be found in the ballot style.
197pub fn map_decoded_ballot_choices_to_decoded_contests(
198    decoded_ballot_choices: &DecodedBallotChoices,
199    contests: &[Contest],
200) -> Result<Vec<DecodedVoteContest>, String> {
201    let mut decoded_contests = vec![];
202
203    for found_contest in contests {
204        let contest_id = &found_contest.id;
205        let found_ballot_choices = decoded_ballot_choices
206            .choices
207            .iter()
208            .find(|ballot_choice| &ballot_choice.contest_id == contest_id)
209            .ok_or_else(|| {
210                format!(
211                    "Can't find contest with id {contest_id} on ballot style"
212                )
213            })?;
214
215        let mut choices = vec![];
216
217        for candidate in &found_contest.candidates {
218            let selected = if found_ballot_choices
219                .choices
220                .iter()
221                .any(|choice| choice.0 == candidate.id)
222            {
223                0
224            } else {
225                -1
226            };
227
228            let decoded_vote_choice = DecodedVoteChoice {
229                id: candidate.id.clone(),
230                selected,
231                write_in_text: None,
232            };
233
234            choices.push(decoded_vote_choice);
235        }
236
237        let decoded_contest = DecodedVoteContest {
238            contest_id: contest_id.clone(),
239            is_explicit_invalid: decoded_ballot_choices.is_explicit_invalid,
240            invalid_errors: found_ballot_choices.invalid_errors.clone(),
241            invalid_alerts: found_ballot_choices.invalid_alerts.clone(),
242            choices,
243        };
244
245        decoded_contests.push(decoded_contest);
246    }
247    Ok(decoded_contests)
248}

250/// Maps an auditable multi-ballot to decoded contests.
251///
252/// # Errors
253/// Returns `Err(String)` if the number of contests does not match or if deserialization fails.
254pub fn map_to_decoded_multi_contest<C: Ctx<P = [u8; 30]>>(
255    ballot: &AuditableMultiBallot,
256) -> Result<Vec<DecodedVoteContest>, String> {
257    let ballot_contests: AuditableMultiBallotContests<C> =
258        ballot.deserialize_contests().map_err(|err| {
259            format!(
260                "Error deserializing auditable multi ballot contest {err:?}"
261            )
262        })?;
263
264    if ballot.config.contests.len() != ballot_contests.contest_ids.len() {
265        return Err(format!(
266            "Invalid number of contests {} != {}",
267            ballot.config.contests.len(),
268            ballot_contests.contest_ids.len()
269        ));
270    }
271
272    let decoded_ballot_choices = BallotChoices::decode_from_30_bytes(
273        &ballot_contests.choice.plaintext,
274        &ballot.config,
275    )
276    .map_err(|err| format!("Error decoding multi ballot plaintext {err:?}"))?;
277
278    let mapped_contests: Vec<Contest> = ballot_contests
279        .contest_ids
280        .clone()
281        .into_iter()
282        .map(|contest_id| -> Result<Contest, String> {
283            ballot
284                .config
285                .contests
286                .clone()
287                .into_iter()
288                .find(|contest_el| contest_el.id == contest_id)
289                .ok_or_else(|| {
290                    format!(
291                        "Can't find contest with id {contest_id} on ballot style"
292                    )
293                })
294        })
295        .collect::<Result<Vec<_>, String>>()?;
296    map_decoded_ballot_choices_to_decoded_contests(
297        &decoded_ballot_choices,
298        &mapped_contests,
299    )
