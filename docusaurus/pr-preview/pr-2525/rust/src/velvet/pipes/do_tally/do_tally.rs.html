62#[instrument]
63pub fn list_tally_sheet_subfolders(path: &Path) -> Vec<PathBuf> {
64    let subfolders = list_subfolders(&path);
65    let tally_sheet_folders: Vec<PathBuf> = subfolders
66        .into_iter()
67        .filter(|path| {
68            let Some(folder_name) = get_folder_name(path) else {
69                return false;
70            };
71            folder_name.starts_with(PREFIX_TALLY_SHEET)
72        })
73        .collect();
74    tally_sheet_folders
75}

77impl DoTally {
78    #[instrument(err, skip_all)]
79    fn save_tally_sheets_breakdown(
80        &self,
81        tally_sheet_results: &Vec<(ContestResult, TallySheet)>,
82        base_file_path: &PathBuf,
83    ) -> Result<()> {
84        let base_breakdown_path = base_file_path.join(OUTPUT_BREAKDOWNS_FOLDER);
85        let mut breakdown_map: HashMap<VotingChannel, ContestResult> = HashMap::new();
86
87        for (contest_result, tally_sheet) in tally_sheet_results {
88            let channel: VotingChannel = tally_sheet.channel.clone().into();
89
90            breakdown_map
91                .entry(channel)
92                .and_modify(|current_result| {
93                    current_result.aggregate(&contest_result, true);
94                })
95                .or_insert_with(|| contest_result.clone());
96        }
97
98        for (channel, contest_result) in breakdown_map {
99            let breakdown_folder_path = base_breakdown_path.join(&channel.to_string());
100            fs::create_dir_all(&breakdown_folder_path)?;
101            let breakdown_file_path = breakdown_folder_path.join(OUTPUT_CONTEST_RESULT_FILE);
102            let contest_result_file = fs::File::create(&breakdown_file_path)?;
103            serde_json::to_writer(contest_result_file, &contest_result)?;
104        }
105
106        Ok(())
107    }
108}

110impl Pipe for DoTally {
111    #[instrument(err, skip_all, name = "DoTally::exec")]
112    fn exec(&self) -> Result<()> {
113        let input_dir_base = self
114            .pipe_inputs
115            .cli
116            .output_dir
117            .as_path()
118            .join(PipeNameOutputDir::DecodeBallots.as_ref());
119        let output_dir_base = self
120            .pipe_inputs
121            .cli
122            .output_dir
123            .as_path()
124            .join(PipeNameOutputDir::DoTally.as_ref());
125        let tally_sheets_dir_base = self.pipe_inputs.root_path_tally_sheets.clone();
126
127        for election_input in &self.pipe_inputs.election_list {
128            // Parallelize the processing of each contest
129            election_input
130                .contest_list
131                .par_iter()
132                .map(|contest_input| {
133                    let input_dir = input_dir_base.clone();
134                    let output_dir = output_dir_base.clone();
135                    let tally_sheets_dir = tally_sheets_dir_base.clone();
136
137                    // These are specific to the contest and need to be cloned for use in area processing.
138                    let election_id_for_contest = contest_input.election_id.clone();
139                    let contest_id_for_contest = contest_input.id.clone();
140                    let contest_object = contest_input.contest.clone();
141                    let contest_op = get_contest_tally_operation(&contest_object);
142
143                    // --- Start of logic for a single contest ---
144                    let _areas_info: Vec<TreeNodeArea> = contest_input // Renamed, original `areas` was unused after info
145                        .area_list
146                        .iter()
147                        .map(|area| (&area.area).into())
148                        .collect();
149                    info!(
150                        "areas for contest {}: {:?}",
151                        contest_id_for_contest, _areas_info
152                    );
153
154                    let areas_tree = Arc::new(
155                        TreeNode::<()>::from_areas(election_input.areas.clone()).map_err(
156                            |err| {
157                                Error::UnexpectedError(format!(
158                                    "Error building area tree for contest {}: {:?}",
159                                    contest_id_for_contest, err
160                                ))
161                            },
162                        )?,
163                    );
164
165                    let census_map: HashMap<String, u64> = contest_input
166                        .area_list
167                        .iter()
168                        .map(|area_input| (area_input.area.id.to_string(), area_input.census))
169                        .collect();
170                    let auditable_votes_map: HashMap<String, u64> = contest_input
171                        .area_list
172                        .iter()
173                        .map(|area_input| {
174                            (area_input.area.id.to_string(), area_input.auditable_votes)
175                        })
176                        .collect();
177
178                    // Parallelize processing for each area within this contest
179                    let area_processing_results: Result<Vec<_>, Error> = contest_input
180                        .area_list
181                        .par_iter()
182                        .map(|area_input| {
183                            // Clone data needed per area task.
184                            let area_id = area_input.id.clone();
185                            let election_id = election_id_for_contest.clone();
186                            let contest_id = contest_id_for_contest.clone();
187
188                            let base_input_path = PipeInputs::build_path(
189                                &input_dir,
190                                &election_id,
191                                Some(&contest_id),
192                                Some(&area_id),
193                            );
194
195                            let base_output_path = PipeInputs::build_path(
196                                &output_dir,
197                                &election_id,
198                                Some(&contest_id),
199                                Some(&area_id),
200                            );
201
202                            let decoded_ballots_file =
203                                base_input_path.join(OUTPUT_DECODED_BALLOTS_FILE);
204
205                            // Create aggregate tally from children areas
206                            let Some(area_tree_node) =
207                                areas_tree.as_ref().find_area(&area_input.id.to_string())
208                            else {
209                                return Err(Error::UnexpectedError(format!(
210                                    "Error finding area {} in areas tree for contest {}",
211                                    area_input.id, contest_id
212                                )));
213                            };
214                            let children_areas = area_tree_node.get_all_children();
215                            let num_children_areas = children_areas
216                                .iter()
217                                .filter(|child| child.id != area_input.id.to_string())
218                                .count();
219
220                            let area_op = get_area_tally_operation(
221                                &election_input.ballot_styles,
222                                contest_object.get_counting_algorithm(),
223                                &area_input.id,
224                            );
225
226                            if num_children_areas > 0usize {
227                                let base_aggregate_path = base_output_path
228                                    .join(OUTPUT_CONTEST_RESULT_AREA_CHILDREN_AGGREGATE_FOLDER);
229                                fs::create_dir_all(&base_aggregate_path)?;
230
231                                let census_size: u64 = children_areas
232                                    .iter()
233                                    .filter_map(|child_area| {
234                                        census_map.get(&child_area.id).copied()
235                                    })
236                                    .sum();
237                                let auditable_votes_size: u64 = children_areas
238                                    .iter()
239                                    .filter_map(|child_area| {
240                                        auditable_votes_map.get(&child_area.id).copied()
241                                    })
242                                    .sum();
243
244                                let children_area_paths: Vec<(PathBuf, Weight)> = children_areas
245                                    .iter()
246                                    .map(|child_area| -> Result<(PathBuf, Weight), Error> {
247                                        let child_area_id = Uuid::parse_str(&child_area.id)
248                                            .map_err(|err| {
249                                                Error::UnexpectedError(format!(
250                                                    "Uuid parse error: {err:?}"
251                                                ))
252                                            })?;
253
254                                        let child_area_weight = get_area_weight(
255                                            &election_input.ballot_styles,
256                                            &child_area_id,
257                                        );
258
259                                        Ok((
260                                            PipeInputs::build_path(
261                                                &input_dir,
262                                                &election_id,
263                                                Some(&contest_id),
264                                                Some(&child_area_id),
265                                            )
266                                            .join(OUTPUT_DECODED_BALLOTS_FILE),
267                                            child_area_weight,
268                                        ))
269                                    })
270                                    .collect::<Result<Vec<(PathBuf, Weight)>, Error>>()?;
271
272                                let counting_algorithm = tally::create_tally(
273                                    &contest_object,
274                                    ScopeOperation::Area(area_op), // The operation of the parent area is used in the aggregate of its children, this makes sense so that each child has the same data available
275                                    children_area_paths,
276                                    census_size,
277                                    auditable_votes_size,
278                                    vec![],
279                                    vec![],
280                                )
281                                .map_err(|e| Error::UnexpectedError(e.to_string()))?;
282                                let res: ContestResult = counting_algorithm
283                                    .tally()
284                                    .map_err(|e| Error::UnexpectedError(e.to_string()))?;
285                                let file_path =
286                                    base_aggregate_path.join(OUTPUT_CONTEST_RESULT_FILE);
287                                let file = fs::File::create(file_path)?;
288                                serde_json::to_writer_pretty(file, &res)?; // Using pretty for readability
289                            }
290
291                            let area_weight =
292                                get_area_weight(&election_input.ballot_styles, &area_input.id);
293
294                            // Create area tally
295                            let counting_algorithm_area = tally::create_tally(
296                                &contest_object,
297                                ScopeOperation::Area(area_op),
298                                vec![(decoded_ballots_file.clone(), area_weight)],
299                                area_input.census,
300                                area_input.auditable_votes,
301                                vec![],
302                                vec![],
303                            )
304                            .map_err(|e| Error::UnexpectedError(e.to_string()))?;
305                            let mut area_tally_results = counting_algorithm_area
306                                .tally()
307                                .map_err(|e| Error::UnexpectedError(e.to_string()))?;
308
309                            if let Some(extended_metrics) =
310                                area_tally_results.extended_metrics.as_mut()
311                            {
312                                extended_metrics.weight = area_weight;
313                            }
314
315                            fs::create_dir_all(&base_output_path)?;
316                            let file_path_area = base_output_path.join(OUTPUT_CONTEST_RESULT_FILE);
317                            let file_area = fs::File::create(file_path_area)?;
318                            serde_json::to_writer_pretty(file_area, &area_tally_results)?; // Using pretty
319
320                            // Tally sheets tally for this area
321                            let mut area_specific_tally_sheet_results: Vec<(
322                                ContestResult,
323                                TallySheet,
324                            )> = vec![];
325                            let input_tally_sheets_dir_path = PipeInputs::build_path(
326                                &tally_sheets_dir,
327                                &election_id,
328                                Some(&contest_id),
329                                Some(&area_id),
330                            );
331
332                            if input_tally_sheets_dir_path.exists()
333                                && input_tally_sheets_dir_path.is_dir()
334                            {
335                                let tally_sheet_folders =
336                                    list_tally_sheet_subfolders(&input_tally_sheets_dir_path);
337                                for tally_sheet_folder in tally_sheet_folders {
338                                    let tally_sheets_file_path =
339                                        tally_sheet_folder.join(INPUT_TALLY_SHEET_FILE);
340                                    let tally_sheet_str = fs::read_to_string(
341                                        &tally_sheets_file_path,
342                                    )
343                                    .map_err(|e| {
344                                        Error::FileAccess(tally_sheets_file_path.to_path_buf(), e)
345                                    })?;
346                                    let tally_sheet: TallySheet =
347                                        serde_json::from_str(&tally_sheet_str)?;
348                                    let output_tally_sheets_folder_path =
349                                        PipeInputs::build_tally_sheet_path(
350                                            &base_output_path,
351                                            &tally_sheet.id, // Assuming TallySheet has an id field
352                                        );
353                                    fs::create_dir_all(&output_tally_sheets_folder_path)?;
354                                    let contest_result_sheet =
355                                        tally::process_tally_sheet(&tally_sheet, &contest_object)
356                                            .map_err(|e| Error::UnexpectedError(e.to_string()))?;
357
358                                    let output_tally_sheets_file_path =
359                                        output_tally_sheets_folder_path
360                                            .join(OUTPUT_CONTEST_RESULT_FILE);
361                                    let contest_result_file_sheet =
362                                        fs::File::create(&output_tally_sheets_file_path)?;
363                                    serde_json::to_writer_pretty(
364                                        contest_result_file_sheet,
365                                        &contest_result_sheet,
366                                    )?; // Using pretty
367                                    area_specific_tally_sheet_results
368                                        .push((contest_result_sheet, tally_sheet));
369                                }
370                            }
371                            // Return data needed for final aggregation for the contest
372                            Ok((
373                                (decoded_ballots_file, area_weight),
374                                area_input.census,
375                                area_input.auditable_votes,
376                                area_specific_tally_sheet_results,
377                                area_tally_results,
378                            ))
379                        })
380                        .collect(); // Collects Result<Vec<(PathBuf, u64, u64, Vec<_>)>, Error>
381
382                    let collected_area_outputs = area_processing_results?; // Propagate error if any area failed
383
384                    // Aggregate results from parallel area processing
385                    let mut contest_ballot_files: Vec<(PathBuf, Weight)> = vec![];
386                    let mut sum_census: u64 = 0;
387                    let mut sum_auditable_votes: u64 = 0;
388                    let mut tally_sheet_results_for_contest: Vec<(ContestResult, TallySheet)> =
389                        vec![];
390                    let mut area_tally_results_for_contest: Vec<ContestResult> = vec![];
391
392                    for (
393                        ballot_file,
394                        census,
395                        auditable_votes_val,
396                        sheet_results,
397                        area_tally_results,
398                    ) in collected_area_outputs
399                    {
400                        contest_ballot_files.push(ballot_file);
401                        sum_census += census;
402                        sum_auditable_votes += auditable_votes_val;
403                        tally_sheet_results_for_contest.extend(sheet_results);
404                        area_tally_results_for_contest.push(area_tally_results);
405                    }
406
407                    // Create contest-level output path (directory for the contest)
408                    let contest_output_dir_path = PipeInputs::build_path(
409                        &output_dir, // This is the output_dir cloned for this contest task
410                        &election_id_for_contest,
411                        Some(&contest_id_for_contest),
412                        None, // No area_id for contest-level summary
413                    );
414                    fs::create_dir_all(&contest_output_dir_path)?; // Ensure contest directory exists
415
416                    self.save_tally_sheets_breakdown(
417                        &tally_sheet_results_for_contest,
418                        &contest_output_dir_path,
419                    )?;
420
421                    let final_only_sheet_results: Vec<ContestResult> =
422                        tally_sheet_results_for_contest
423                            .iter()
424                            .map(|(res, _)| res.clone())
425                            .collect();
426
427                    // Create final contest tally
428                    let final_counting_algorithm = tally::create_tally(
429                        &contest_object,
430                        ScopeOperation::Contest(contest_op),
431                        contest_ballot_files,
432                        sum_census,
433                        sum_auditable_votes,
434                        final_only_sheet_results,
435                        area_tally_results_for_contest,
436                    )
437                    .map_err(|e| Error::UnexpectedError(e.to_string()))?;
438                    let final_res = final_counting_algorithm
439                        .tally()
440                        .map_err(|e| Error::UnexpectedError(e.to_string()))?;
441
442                    let final_contest_result_file_path =
443                        contest_output_dir_path.join(OUTPUT_CONTEST_RESULT_FILE);
444                    let final_file = fs::File::create(final_contest_result_file_path)?;
445                    serde_json::to_writer_pretty(final_file, &final_res)?; // Using pretty
446
447                    Ok(()) // Result for this contest's processing
448                })
449                .collect::<Result<Vec<()>, Error>>()?; // Collect results from parallel contest processing
450        }
451        Ok(())
452    }
453}

477    pub under_votes: u64,
478    // Total actual marks count of candidates in the contest. Only counted UV and fully votes.

482    // votes possible in the contest)
483    pub expected_votes: u64,
484    //Total counted ballots
485    pub total_ballots: u64,
486    pub weight: Weight, // Used to store the actual weight used to tally an specific area.
487    pub total_weight: u64, // Used to calculate the right percentage_votes in aggregate
488}

491    #[instrument(skip_all)]

494        result.over_votes += other.over_votes;
495        result.under_votes += other.under_votes;
496        result.votes_actually += other.votes_actually;
497        result.expected_votes += other.expected_votes;
498        result.total_ballots += other.total_ballots;
499        result.total_weight += other.total_weight;
500        result
501    }

534impl ContestResult {
535    #[instrument(skip_all)]
536    pub fn calculate_percentages(&self) -> ContestResult {
537        let total_weight = self
538            .extended_metrics
539            .clone()
540            .unwrap_or_default()
541            .total_weight;
542        let candidate_result: Vec<CandidateResult> = self
543            .candidate_result
544            .clone()
545            .into_iter()
546            .map(|candidate_result| {
547                let percentage_votes = (candidate_result.total_count as f64
548                    / cmp::max(1, total_weight) as f64)
549                    * 100.0;
550                let mut new_candidate_result = candidate_result.clone();
551                new_candidate_result.percentage_votes = percentage_votes.clamp(0.0, 100.0);
552
553                new_candidate_result
554            })
555            .collect();
556        let total_votes = self.total_votes;
557        let total_votes_base = cmp::max(1, total_votes) as f64;
558        let count_valid = self.total_valid_votes;
559
560        let census_base = cmp::max(1, self.census) as f64;
561
562        // `percentage_auditable_votes` is calculated over `census_base`.
563        // Otherwise we could end up with strange percentages. Imagine a test
564        // election with 2 auditable votes and 1 valid vote. That's maybe 66%
565        // auditable votes over the census, but 200% over total votes.
566        let percentage_auditable_votes = (self.auditable_votes as f64) * 100.0 / census_base;
567        let percentage_total_votes = (total_votes as f64) * 100.0 / census_base;
568        let percentage_total_valid_votes = (count_valid as f64 * 100.0) / total_votes_base;
569        let percentage_total_invalid_votes =
570            (self.total_invalid_votes as f64 * 100.0) / total_votes_base;
571        let percentage_total_blank_votes =
572            (self.total_blank_votes as f64 * 100.0) / total_votes_base;
573        let percentage_invalid_votes_explicit =
574            (self.invalid_votes.explicit as f64 * 100.0) / total_votes_base;
575        let percentage_invalid_votes_implicit =
576            (self.invalid_votes.implicit as f64 * 100.0) / total_votes_base;
577
578        let mut contest_result = self.clone();
579        contest_result.percentage_census = 100.0;
580        contest_result.percentage_auditable_votes = percentage_auditable_votes.clamp(0.0, 100.0);
581        contest_result.percentage_total_votes = percentage_total_votes.clamp(0.0, 100.0);
582        contest_result.percentage_total_valid_votes =
583            percentage_total_valid_votes.clamp(0.0, 100.0);
584        contest_result.percentage_total_invalid_votes =
585            percentage_total_invalid_votes.clamp(0.0, 100.0);
586        contest_result.percentage_total_blank_votes =
587            percentage_total_blank_votes.clamp(0.0, 100.0);
588        contest_result.percentage_invalid_votes_explicit =
589            percentage_invalid_votes_explicit.clamp(0.0, 100.0);
590        contest_result.percentage_invalid_votes_implicit =
591            percentage_invalid_votes_implicit.clamp(0.0, 100.0);
592        contest_result.candidate_result = candidate_result;
593        contest_result
594    }
595
596    #[instrument(skip_all)]
597    pub fn aggregate(&self, other: &ContestResult, add_census: bool) -> ContestResult {
598        let mut aggregate = self.clone();
599        if add_census {
600            aggregate.census += other.census;
601        }
602        let aggregate_metrics = aggregate.extended_metrics.unwrap_or_default();
603        aggregate.extended_metrics =
604            Some(aggregate_metrics.aggregate(&other.extended_metrics.clone().unwrap_or_default()));
605        aggregate.total_votes += other.total_votes;
606        aggregate.total_valid_votes += other.total_valid_votes;
607        aggregate.total_invalid_votes += other.total_invalid_votes;
608        aggregate.total_blank_votes += other.total_blank_votes;
609        aggregate.invalid_votes = aggregate.invalid_votes.aggregate(&other.invalid_votes);
610
611        let mut candidate_map: HashMap<String, CandidateResult> = HashMap::new();
612
613        for candidate_result in &self.candidate_result {
614            candidate_map.insert(
615                candidate_result.candidate.id.clone(),
616                candidate_result.clone(),
617            );
618        }
619
620        for candidate_result in &other.candidate_result {
621            candidate_map
622                .entry(candidate_result.candidate.id.clone())
623                .and_modify(|entry| entry.total_count += candidate_result.total_count)
624                .or_insert_with(|| candidate_result.clone());
625        }
626
627        aggregate.candidate_result = candidate_map.into_values().collect();
628
629        aggregate.calculate_percentages()
630    }
631}

636    pub percentage_votes: f64,
637    pub total_count: u64,
