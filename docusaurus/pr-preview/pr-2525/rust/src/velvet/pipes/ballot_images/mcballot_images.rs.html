51#[instrument(skip_all)]
52pub fn qr_encode_choices(contests: &Vec<ContestData>, title: &str) -> String {
53    let is_blank: bool = contests.iter().all(|contest| contest.is_blank());
54    let mut data = vec![title.to_string()];
55    if is_blank {
56        data.push("ABSTENTION".to_string());
57    } else {
58        for contest in contests {
59            data.push(contest.contest.name.clone().unwrap_or_default());
60            for candidate in &contest.decoded_choices {
61                if !candidate.is_selected() {
62                    continue;
63                }
64                let candidate_name = candidate
65                    .candidate
66                    .clone()
67                    .map(|cand| cand.name)
68                    .flatten()
69                    .unwrap_or_default();
70                data.push(candidate_name);
71            }
72        }
73    }
74    data.join(":")
75}

77#[instrument(skip_all)]
78fn sort_candidates(candidates: &mut Vec<DecodedChoice>, order_field: CandidatesOrder) {
79    match order_field {
80        CandidatesOrder::Alphabetical => candidates.sort_by(|a, b| {
81            let name_a = match &a.candidate {
82                Some(candidate) => candidate
83                    .alias
84                    .as_ref()
85                    .or(candidate.name.as_ref())
86                    .unwrap_or(&String::new())
87                    .to_lowercase(),
88                None => String::new(),
89            };
90
91            let name_b = match &b.candidate {
92                Some(candidate) => candidate
93                    .alias
94                    .as_ref()
95                    .or(candidate.name.as_ref())
96                    .unwrap_or(&String::new())
97                    .to_lowercase(),
98                None => String::new(),
99            };
100
101            name_a.cmp(&name_b)
102        }),
103        CandidatesOrder::Custom => {
104            candidates.sort_by(|a, b| {
105                let sort_order_a = match &a.candidate {
106                    Some(candidate) => candidate
107                        .presentation
108                        .as_ref()
109                        .and_then(|p| p.sort_order)
110                        .unwrap_or(-1),
111                    None => -1, // Default value when `a.candidate` is `None`
112                };
113
114                let sort_order_b = match &b.candidate {
115                    Some(candidate) => candidate
116                        .presentation
117                        .as_ref()
118                        .and_then(|p| p.sort_order)
119                        .unwrap_or(-1),
120                    None => -1, // Default value when `b.candidate` is `None`
121                };
122
123                sort_order_a.cmp(&sort_order_b)
124            })
125        }
126
127        CandidatesOrder::Random => {
128            // We don't randomize in results
129        }
130    }
131}

139    #[instrument(skip_all, err)]
140    fn print_ballot_images(
141        &self,
142        ballots: &[Bridge],
143        contests: &Vec<Contest>,
144        election_input: &InputElectionConfig,
145        pipe_config: &PipeConfigBallotImages,
146        area_name: &str,
147    ) -> Result<(Option<Vec<u8>>, Vec<u8>)> {
148        // 1. Gather the sign_data for all ballots/contests
149        let mut bulk_sign_requests = Vec::new();
150
151        // We'll store some structures that map from (ballotIndex, contestIndex, pageNum)
152        // to the sign_data string, so later we can fill in the signatures.
153        struct ContestLocator {
154            sign_id: String,
155            ballot_index: usize,
156            contest_index: usize,
157        }
158        let mut locators = Vec::new();
159
160        let contest_map: HashMap<String, Contest> = contests
161            .iter()
162            .map(|c| (c.id.to_string(), c.clone()))
163            .collect();
164        let execution_annotations: Option<HashMap<String, String>> =
165            pipe_config.execution_annotations.clone();
166        let precint_id = election_input
167            .annotations
168            .get(&"miru:precinct-code".to_string())
169            .map(|s| s.as_str())
170            .unwrap_or_default();
171        let mut page_number = 1;
172        let election_event_id = election_input
173            .election_event_annotations
174            .get("miru:election-event-id")
175            .map(|s| s.as_str())
176            .unwrap_or_default();
177        let election_id = election_input
178            .annotations
179            .get("miru:election-id")
180            .map(|s| s.as_str())
181            .unwrap_or_default();
182
183        let mut ballot_data = vec![];
184        for (b_idx, ballot) in ballots.iter().enumerate() {
185            let mut cds = vec![];
186            for (c_idx, contest_choices) in ballot.choices.iter().enumerate() {
187                let contest = contest_map
188                    .get(&contest_choices.contest_id)
189                    .ok_or_else(|| Error::UnexpectedError("Can't get contest".into()))?;
190
191                let mut choices = DecodedChoice::from_dvcs(contest_choices, contest);
192
193                let candidates_order = contest
194                    .presentation
195                    .clone()
196                    .unwrap_or_default()
197                    .candidates_order
198                    .unwrap_or_default();
199                sort_candidates(&mut choices, candidates_order.clone());
200
201                let num_selected = choices.iter().filter(|can| can.is_selected()).count();
202                let undervotes = contest.max_votes - (num_selected as i64);
203                let overvotes = if (num_selected as i64) > contest.max_votes {
204                    (num_selected as i64) - contest.max_votes
205                } else {
206                    0
207                };
208
209                // Instead of calling ecies_sign_data here, we only CREATE the data
210                let (digital_signature, sign_data) = if pipe_config.acm_key.is_some() {
211                    let data_str = format!(
212                        "{}:{}:{}:{}:{}",
213                        election_event_id,
214                        precint_id,
215                        ballot.mcballot.serial_number.clone().unwrap_or_default(),
216                        election_id,
217                        page_number.to_string()
218                    );
219                    // We'll push this into our bulk_sign_requests
220                    // We also need a unique ID to correlate the signature
221                    let sign_id = format!("b{}_c{}_p{}", b_idx, c_idx, page_number);
222
223                    bulk_sign_requests.push(SignRequest {
224                        id: sign_id.clone(),
225                        data: data_str.clone(),
226                    });
227
228                    // We'll store so we can insert the signature after we do the bulk sign
229                    locators.push(ContestLocator {
230                        sign_id: sign_id.clone(),
231                        ballot_index: b_idx,
232                        contest_index: c_idx,
233                    });
234
235                    // We do not have a signature yet, so just placeholders
236                    (None, Some(data_str))
237                } else {
238                    (None, None)
239                };
240
241                let cd: ContestData = ContestData {
242                    contest: contest.clone(),
243                    decoded_choices: choices,
244                    undervotes,
245                    overvotes,
246                    digital_signature,
247                    sign_data,
248                    page_number: Some(page_number),
249                };
250
251                page_number += 1;
252                cds.push(cd);
253            }
254
255            cds.sort_by(|a, b| b.contest.name.cmp(&a.contest.name));
256
257            let title = pipe_config.extra_data["title"]
258                .as_str()
259                .map(|val| val.to_string())
260                .unwrap_or(DEFAULT_MCBALLOT_TITLE.to_string());
261            let encoded_vote = qr_encode_choices(&cds, &title);
262            let is_blank = cds.iter().all(|choice| choice.is_blank());
263
264            let bd = BallotData {
265                id: ballot.mcballot.serial_number.clone().unwrap_or_default(),
266                encoded_vote,
267                is_invalid: ballot.mcballot.is_explicit_invalid,
268                is_blank,
269                contest_choices: cds,
270            };
271
272            ballot_data.push(bd);
273            page_number += 1; // inc by one for summary page
274        }
275
276        // 2. Now we do exactly one bulk sign if we have any sign_data
277        let mut signatures_map: HashMap<String, String> = HashMap::new();
278        if let Some(acm_key) = &pipe_config.acm_key {
279            if !bulk_sign_requests.is_empty() {
280                signatures_map = ecies_sign_data_bulk(acm_key, &bulk_sign_requests)
281                    .map_err(|e| Error::UnexpectedError(format!("Error in bulk signing: {}", e)))?;
282            }
283        }
284
285        // 3. Use the `locators` array to stitch the signatures back into `ballot_data`
286        for locator in locators {
287            // get the actual signature from the map
288            if let Some(sig_base64) = signatures_map.get(&locator.sign_id) {
289                if ballot_data.len() <= locator.ballot_index {
290                    return Err(Error::UnexpectedError(format!(
291                        "index out of bounds for ballot_index {} and length {}",
292                        locator.ballot_index,
293                        ballot_data.len()
294                    )));
295                }
296                let bd = &mut ballot_data[locator.ballot_index];
297                if bd.contest_choices.len() <= locator.contest_index {
298                    return Err(Error::UnexpectedError(format!(
299                        "index out of bounds for contest_index {} and length {}",
300                        locator.contest_index,
301                        bd.contest_choices.len()
302                    )));
303                }
304                let cd = &mut bd.contest_choices[locator.contest_index];
305
306                cd.digital_signature = Some(sig_base64.clone());
307            }
308        }
309
310        let td = TemplateData {
311            election_name: election_input.name.clone(),
312            election_alias: election_input.alias.clone(),
313            ballot_data,
314            area: area_name.to_string(),
315            election_annotations: election_input.annotations.clone(),
316            election_dates: election_input.dates.clone(),
317            execution_annotations: execution_annotations.unwrap_or_default().clone(),
318        };
319
320        let mut map = Map::new();
321        map.insert("data".to_string(), serde_json::to_value(&td)?);
322        map.insert(
323            "extra_data".to_string(),
324            serde_json::to_value(&pipe_config.extra_data)?,
325        );
326
327        let rendered_user_template = reports::render_template_text(&pipe_config.template, map)
328            .map_err(|e| {
329                Error::UnexpectedError(format!(
330                    "Error during render_template_text from report.hbs template file: {}",
331                    e
332                ))
333            })?;
334
335        let mut system_map = Map::new();
336        system_map.insert(
337            "rendered_user_template".to_string(),
338            serde_json::to_value(&rendered_user_template)?,
339        );
340
341        if let serde_json::Value::Object(obj) = &pipe_config.extra_data {
342            for (key, value) in obj {
343                system_map.insert(key.clone(), value.clone());
344            }
345        }
346
347        let bytes_html = reports::render_template_text(&pipe_config.system_template, system_map)
348            .map_err(|e| {
349                Error::UnexpectedError(format!(
350                    "Error during render_template_text from report.hbs template file: {}",
351                    e
352                ))
353            })?;
354
355        let pdf_options = match pipe_config.pdf_options.clone() {
356            Some(options) => Some(options.to_print_to_pdf_options()),
357            None => None,
358        };
359
360        let bytes_pdf = if pipe_config.enable_pdfs {
361            Some(
362                pdf::sync::PdfRenderer::render_pdf(bytes_html.clone(), pdf_options).map_err(
363                    |e| Error::UnexpectedError(format!("Error during PDF rendering: {}", e)),
364                )?,
365            )
366        } else {
367            None
368        };
369
370        Ok((bytes_pdf, bytes_html.into_bytes()))
371    }

396#[instrument(err, skip_all)]
397fn generate_hashed_filename(
398    path: &PathBuf,
399    name: &str,
400    hash_bytes: &[u8],
401    area_id: &str,
402    election_input: &InputElectionConfig,
403    from_ballot: Option<&Bridge>,
404    to_ballot: Option<&Bridge>,
405) -> Result<PathBuf> {
406    let path = path.as_path();
407    let country_code = election_input
408        .areas
409        .iter()
410        .find(|area| area.id == area_id.to_string())
411        .and_then(|area| {
412            area.annotations
413                .as_ref()
414                .and_then(|annotations| annotations.get("miru:area-station-id"))
415                .and_then(|value| value.as_str())
416        })
417        .unwrap_or("");
418    let post_code = election_input
419        .annotations
420        .get("miru:precinct-code")
421        .map(|s| s.as_str())
422        .unwrap_or("");
423    let clustered_precint_id = election_input
424        .annotations
425        .get("clustered_precint_id")
426        .map(|s| s.as_str())
427        .unwrap_or("");
428
429    let from_ballot_id = match from_ballot {
430        Some(from_ballot) => from_ballot.mcballot.serial_number.as_deref().unwrap_or(""),
431        None => "000000000",
432    };
433    let to_ballot_id = match to_ballot {
434        Some(to_ballot) => to_ballot.mcballot.serial_number.as_deref().unwrap_or(""),
435        None => "000000000",
436    };
437
438    let hash_hex = hex::encode(hash_bytes);
439
440    let new_filename = format!(
441        "{name}_{post_code}_{country_code}_{clustered_precint_id}_{from_ballot_id}-{to_ballot_id}_{hash_hex}.pdf"
442    );
443
444    Ok(path.join(new_filename))
445}

447#[derive(Serialize, Debug, Clone)]
448struct BallotCsvData {
449    pub file_name: String,
450    pub hash: String,
451}
452impl Pipe for MCBallotImages {
453    #[instrument(err, skip_all, name = "MultiBallotReceipts::exec")]
454    fn exec(&self) -> Result<()> {
455        let pipe_config: PipeConfigBallotImages = self.get_config()?;
456        let pipe_data = get_pipe_data();
457        for election_input in &self.pipe_inputs.election_list {
458            let area_contests_map = election_input.get_area_contest_map();
459
460            let files = Mutex::new(vec![]);
461
462            for (area_id, area_contests) in area_contests_map {
463                let path_ballots = PipeInputs::mcballots_path(
464                    &self
465                        .pipe_inputs
466                        .cli
467                        .output_dir
468                        .join(PipeNameOutputDir::DecodeMCBallots.as_ref())
469                        .as_path(),
470                    &election_input.id,
471                    &area_id,
472                )
473                .join(OUTPUT_DECODED_BALLOTS_FILE);
474
475                if path_ballots.exists() {
476                    let f = fs::File::open(path_ballots.as_path())
477                        .map_err(|e| Error::FileAccess(path_ballots.as_path().to_path_buf(), e))?;
478                    let mcballots: Vec<DecodedBallotChoices> = crate::utils::parse_file(f)?;
479
480                    let ballots = convert_ballots(election_input, mcballots)?;
481                    let report_options = pipe_config.report_options.clone().unwrap_or_default();
482                    let max_threads = report_options.max_threads.unwrap_or_else(|| 3);
483                    let pool = ThreadPoolBuilder::new()
484                        .num_threads(max_threads)
485                        .build()
486                        .map_err(|e| {
487                            Error::UnexpectedError(format!("Error building thread pool: {}", e))
488                        })?;
489
490                    let max_items_per_report =
491                        report_options.max_items_per_report.unwrap_or_else(|| 100);
492
493                    let path = PipeInputs::mcballots_path(
494                        &self
495                            .pipe_inputs
496                            .cli
497                            .output_dir
498                            .join(&pipe_data.pipe_name_output_dir)
499                            .as_path(),
500                        &election_input.id,
501                        &area_id,
502                    );
503
504                    let chunks: Vec<&[Bridge]> = match ballots.is_empty() {
505                        true => vec![&[] as &[Bridge]],
506                        false => {
507                            info!("ballots len = {len}", len = ballots.len());
508                            ballots.chunks(max_items_per_report).collect()
509                        }
510                    };
511
512                    let result: Result<(), Error> = pool.install(|| {
513                        chunks.into_par_iter().enumerate().try_for_each(
514                            |(chunk_index, chunk)| {
515                                info!(
516                                    "processing batch {chunk_index} len = {len}",
517                                    len = chunk.len()
518                                );
519                                let (bytes_pdf, bytes_html) = self.print_ballot_images(
520                                    chunk,
521                                    &area_contests.contests,
522                                    &election_input,
523                                    &pipe_config,
524                                    &area_contests.area_name,
525                                )?;
526
527                                fs::create_dir_all(&path)?;
528
529                                if let Some(ref some_bytes_pdf) = bytes_pdf {
530                                    // pdf file creation
531                                    let pdf_hash =
532                                        hash_sha256(some_bytes_pdf.as_slice()).map_err(|e| {
533                                            Error::UnexpectedError(format!(
534                                                "Error during hash pdf bytes: {}",
535                                                e
536                                            ))
537                                        })?;
538
539                                    let base_file_name = pipe_data.output_file.clone();
540                                    let from_ballot = match ballots.is_empty() {
541                                        true => None,
542                                        false => Some(chunk.first().ok_or(
543                                            Error::UnexpectedError("Can't get first chunk".into()),
544                                        )?),
545                                    };
546
547                                    let to_ballot = match ballots.is_empty() {
548                                        true => None,
549                                        false => Some(chunk.last().ok_or(
550                                            Error::UnexpectedError("Can't get last chunk".into()),
551                                        )?),
552                                    };
553
554                                    let file = generate_hashed_filename(
555                                        &path,
556                                        &base_file_name.clone(),
557                                        &pdf_hash,
558                                        &area_id.to_string(),
559                                        election_input,
560                                        from_ballot,
561                                        to_ballot,
562                                    )
563                                    .map_err(|e| {
564                                        Error::UnexpectedError(format!(
565                                            "Error during hash pdf bytes: {}",
566                                            e
567                                        ))
568                                    })?;
569
570                                    let file_name = file
571                                        .file_name()
572                                        .ok_or(Error::UnexpectedError(
573                                            "Can't get file name".into(),
574                                        ))?
575                                        .to_str()
576                                        .ok_or(Error::UnexpectedError(
577                                            "Can't get file name".into(),
578                                        ))?;
579                                    let bytes_json = file_name.as_bytes().to_vec();
580                                    let file_hash = hash_b64(&bytes_json).map_err(|err| {
581                                        Error::UnexpectedError(format!(
582                                            "Error hashing the results file: {err:?}"
583                                        ))
584                                    })?;
585
586                                    // Lock the mutex before modifying the vector
587                                    let mut files_lock = files.lock().map_err(|e| {
588                                        Error::UnexpectedError(format!(
589                                            "Error locking files: {}",
590                                            e
591                                        ))
592                                    })?;
593                                    files_lock.push(BallotCsvData {
594                                        file_name: file_name.to_string(),
595                                        hash: file_hash,
596                                    });
597
598                                    let mut file = OpenOptions::new()
599                                        .write(true)
600                                        .truncate(true)
601                                        .create(true)
602                                        .open(file)?;
603                                    file.write_all(some_bytes_pdf)?;
604                                }
605
606                                let file = path.join(format!(
607                                    "{}_batch-{}.html",
608                                    pipe_data.output_file, chunk_index,
609                                ));
610
611                                let mut file = OpenOptions::new()
612                                    .write(true)
613                                    .truncate(true)
614                                    .create(true)
615                                    .open(file)?;
616                                file.write_all(&bytes_html)?;
617                                Ok::<(), Error>(())
618                            },
619                        )?;
620
621                        // Write the CSV file of file names and hashes ONLY for `ballot` type
622                        if pipe_data.output_file.clone() == BALLOT_IMAGES_OUTPUT_FILE {
623                            let csv_filename = format!("ballots_files.csv");
624                            let csv_path = path.join(csv_filename);
625                            let files_lock = files.lock().map_err(|e| {
626                                Error::UnexpectedError(format!("Error locking files: {}", e))
627                            })?;
628
629                            let rt = Runtime::new()?;
630                            rt.block_on(async {
631                                write_file_hash_csv(files_lock.clone(), csv_path)
632                                    .await
633                                    .map_err(|e| {
634                                        Error::UnexpectedError(format!(
635                                            "Error writing file hash CSV: {}",
636                                            e
637                                        ))
638                                    })
639                            })?;
640                        }
641
642                        Ok(())
643                    });
644
645                    if let Err(e) = result {
646                        eprintln!("Error processing: {}", e);
647                    }
648                } else {
649                    println!(
650                        "[{}] File not found: {} -- Not processed",
651                        &pipe_data.pipe_name,
652                        path_ballots.display()
653                    );
654                };
655            }
656        }
657
658        Ok(())
659    }
660}

662#[derive(Serialize, Debug)]
663pub struct TemplateData {
664    pub ballot_data: Vec<BallotData>,

701#[derive(Serialize, Debug)]
702struct DecodedChoice {
703    pub choice: DecodedVoteChoice,
704    pub candidate: Option<Candidate>,
705}
706impl DecodedChoice {
707    pub fn is_selected(&self) -> bool {
708        self.choice.is_selected()
709    }
710    fn from_dvcs(dvc: &DecodedVoteContest, contest: &Contest) -> Vec<Self> {
711        dvc.choices
712            .iter()
713            .map(|choice| DecodedChoice {
714                choice: choice.clone(),
715                candidate: contest
716                    .candidates
717                    .iter()
718                    .find(|c| c.id == choice.id)
719                    .cloned(),
720            })
721            .collect::<Vec<DecodedChoice>>()
722    }

736// We are reusing some functionality from the standard receipts pipe/template,
737// so it helps to convert mcballots to dcv format
738#[instrument(err, skip_all)]
739fn convert_ballots(
740    election_input: &InputElectionConfig,
741    mcballots: Vec<DecodedBallotChoices>,
742) -> Result<Vec<Bridge>> {
743    let mut ret = vec![];
744
745    let contest_dvc_map = crate::utils::get_contest_dvc_map(election_input);
746
747    for dbc in mcballots {
748        let mut ballot_dvcs = vec![];
749        for contest in &dbc.choices {
750            let blank: Option<&HashMap<String, DecodedVoteChoice>> =
751                contest_dvc_map.get(&contest.contest_id);
752            if let Some(blank) = blank {
753                let mut next = blank.clone();
754                for choice in &contest.choices {
755                    let blank = next.get(&choice.0);
756                    if let Some(blank) = blank {
757                        let mut marked = blank.clone();
758                        marked.selected = 1;
759                        next.insert(choice.0.clone(), marked);
760                    } else {
761                        return Err(Error::UnexpectedError(format!(
762                            "could not find candidate for choice"
763                        )));
764                    }
765                }
766                let mut values: Vec<DecodedVoteChoice> = next.into_values().collect();
767                values.sort_by(|a, b| a.id.cmp(&b.id));
768
769                let marked_contest = DecodedVoteContest {
770                    contest_id: contest.contest_id.clone(),
771                    is_explicit_invalid: dbc.is_explicit_invalid,
772                    // FIXME
773                    invalid_alerts: vec![],
774                    // FIXME
775                    invalid_errors: vec![],
776                    choices: values,
777                };
778                ballot_dvcs.push(marked_contest);
779            } else {
780                return Err(Error::UnexpectedError(format!(
781                    "could not find choices for contest"
782                )));
783            }
784        }
785        ret.push(Bridge::new(dbc, ballot_dvcs));
786    }
787
788    Ok(ret)
789}

791pub async fn write_file_hash_csv(data: Vec<BallotCsvData>, path: PathBuf) -> Result<()> {
792    let headers = vec!["file_name".to_string(), "hash".to_string()];
793
794    let mut writer = Writer::from_writer(vec![]);
795
796    writer.write_record(&headers).map_err(|e| {
797        Error::UnexpectedError(format!("Failed to write headers to CSV file: {}", e))
798    })?;
799
800    for entry in data {
801        writer
802            .write_record(&[entry.file_name, entry.hash])
803            .map_err(|e| Error::UnexpectedError(format!("Failed to write record: {}", e)))?;
804    }
805
806    let data_bytes = writer
807        .into_inner()
808        .map_err(|e| Error::UnexpectedError(format!("Failed to flush CSV writer: {}", e)))?;
809
810    let mut file = OpenOptions::new()
811        .write(true)
812        .truncate(true)
813        .create(true)
814        .open(path)?;
815    file.write_all(&data_bytes)?;
816
817    Ok(())
818}
